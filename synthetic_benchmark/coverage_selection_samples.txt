བྱང་ཆུབ་ཀྱི་སེམས་སྤེལ་ཞིང་སྦྱང་བ་དང་། སེམས་ཅན་བློས་མི་བཏང་བ་དང་། ནག་པོའི་ཆོས་བཞི་སྤངས་ནས་དཀར་པོའི་ཆོས་སྤེལ་བའོ། དང་པོ་བསོད་ནམས་ཀྱི་ཚོགས་རྒྱ་ཆེན་པོ་བསག་པ་ནི། ཧཱི༽ དེ་ལྟར་བྱང་ཆུབ་སོགས། གཉིས་པ་ཕན་ཡོན་མཐོང་ཞིང་རྗེས་སུ་དྲན་པར་བྱ་བ་ནི། ཨཱི༽ དེ་ཡང་དུས་ཐམས་ཅད་དུ་སོགས། གསུམ་པ་བྱང་ཆུབ་ཀྱི་སེམས་སྤེལ་ཞིང་སྦྱང་བ་ནི། ཀཱུ༽ ཇི་ལྟར་སློབ་སོགས། བཞི་པ་སེམས་ཅན་བློས་མི་བཏང་བ་ནི། ཁཱུ༽ དེ་ཡང་སེམས་ཅན་སོགས། ལྔ་པ་ནག་པོའི་ཆོས་བཞི་སྤངས་ནས་དཀར་པོའི་ཆོས་སྤེལ་བ་ནི། གཱུ༽ དེ་ཐམས་ཅད་དུའང་ཆོས་བཞི་སོགས། ལྔ་པོ་དེས་སྨོན་པའི་བསླབ་བྱ་སོང་། གཉིས་པ་འཇུག་པའི་བསླབ་བྱ་ལའང་གཉིས་ཏེ། སྡོམ་པ་གོ་རིམ་ངེས་པའི་ཚུལ་གྱིས་མདོར་བསྟན་པ་དང་། རྒྱས་པར་བཤད་པའོ། དང་པོ་མདོར་བསྟན་པ་ནི། ངཱུ༽ འཇུག་པའི་སོགས། གཉིས་པ་རྒྱས་བཤད་ལའང་གསུམ་སྟེ། སྡོམ་པའི་ཚུལ་ཁྲིམས་དང་། སེམས་ཅན་དོན་བྱེད་ཀྱི་ཚུལ་ཁྲིམས་དང་། དགེ་བ་ཆོས་སྡུད་ཀྱི་ཚུལ་ཁྲིམས་གསུམ་བསྟན་པའོ། དང་པོ་ལའང་གསུམ་སྟེ། སྡོམ་པའི་ཚུལ་ཁྲིམས་བསྲུང་དགོས་པའི་འཐད་པ་དགོད་པ། རྩ་བའི་ལྟུང་བ་བརྒྱད་བསྟན་པ། ཡན་ལག་གི་ལྟུང་བ་སུམ་ཅུ་རྩ་གཅིག་བསྟན་པའོ། དང་པོ་སྡོམ་པའི་ཚུལ་ཁྲིམས་བསྲུང་དགོས་པའི་འཐད་པ་དགོད་པ་ནི། ཅཱུ༽ དེ་ལ་སྤྱིར་སོགས། གཉིས་པ་རྩ་བའི་ལྟུང་བ་བརྒྱད་བསྟན་པ་ནི། ཆཱུ༽ དེ་ལ་བྱེ་བྲག་སོགས། གསུམ་པ་ཡན་ལག་གི་ལྟུང་བ་སུམ་ཅུ་རྩ་གཅིག་བསྟན་པ་ནི། ཇཱུ༽ དེ་རྣམས་ཀྱི་སོགས། གསུམ་པོ་དེས་སྡོམ་པའི་ཚུལ་ཁྲིམས་བསྟན་པ་སོང་༑ གཉིས་པ་སེམས་ཅན་དོན་བྱེད་ཀྱི་ཚུལ་ཁྲིམས་ལའང་བཞི་སྟེ། སེམས་ཅན་དོན་བྱེད་ཀྱི་ཚུལ་ཁྲིམས་དངོས་དང་། གཏོང་ཐོབ་དང་། ཉམས་ན་ཕྱིར་བཅོས་དང་།

གློ་བུར་གཟུང་འཛིན་སྦུབས་རྒྱ་ལས་གྲོལ་ཚེ། ཆོས་གཟུགས་ཡེ་ཤེས་ཉི་མ་ཁོང་ནས་འཆར། ཡིད་སྨོན་བྲལ་བ་འབྲས་བུའི་རྩེ་མོ་ལགས། ཚུལ་འདི་དཔལ་ལྡན་བླ་མའི་ཞལ་གྱི་རྒྱུན། ཅུང་ཟད་སྙིང་གི་བུམ་པར་འཁྱིལ་བའི་ཆ། གཡོ་སྒྱུ་མེད་པས་ཕྱོགས་ཙམ་ཞུས་པ་ལགས། འདི་དོན་ཐུགས་ལ་འཐད་ན་དེ་བཞིན་མཛོད། མི་འཐད་འགལ་བར་གཟིགས་ན་མཐོལ་ལོ་བཤགས། མཉམ་ཉིད་ཀློང་དུ་དགོངས་པར་མཛད་དུ་གསོལ། དེ་སྐད་ཅོལ་ཆུང་པདྨ་བི་ཛ་ཡས། གང་དྲན་འཕྲལ་མར་ཕུལ་ལོ་དགེ་གྱུར་ཅིག ། ཅེས་པའང་མཁས་བཙུན་བཟང་ཡོན་ཕུལ་དུ་བྱུང་བའི་གོ་ཆེ་མཆོག་སྤྲུལ་ཀརྨ་བཤད་སྒྲུབ་ཆོས་ཀྱི་ཉི་མའི་ཐུགས་བཞེད་བཞིན། འོག་མིན་བླ་མའི་གདན་ས་ཙཱ་འདྲ་རིན་ཆེན་བྲག་གི་གཟིམས་སྤྱིལ་ནས་གང་ཤར་བཅོས་མེད་དུ་བྲིས་ཏེ་ཕུལ་བ་དགེ། དུས༵་གསུམ༵་གཤེ༵གས་པ༵འི་རྒྱལ༵་བ༵་སྲས༵་དང༵་བཅ༵ས། གང་ལས་གཤེགས་པའི་ལམ་མཆོག་བླ་མེད་ཕུལ། རིན་ཆེན་བྱང་ཆུབ་སེམས་ལ་ཕྱག་བྱས་ནས། ཕན་པའི་སྙིང་གཏམ་སྨྲ་ཡི་སྙིང་ལ་ཆོངས། ཡང་ཡང་རྙེད་དཀའ་ཤིན་ཏུ་དོན་ཆེ་བའི། ཡིད་བཞིན་ནོར་ལས་ཁྱད་འཕགས་དལ་འབྱོར་རྟེན། སྔོན་བསགས་མཐུ་ཡིས་ལན་གཅིག་རྙེད་དུས་འདིར། ཤིན་ཏུ་བརྩོན་པས་སྙིང་ནས་ཆོས་མཛོད་ཅིག ། སྣོད་བཅུད་གང་ལ་བསམས་ཀྱང་རྟག་པ་མེད༑ ༑ཁྱད་པར་སྐྱེས་བུའི་ཚེ་སྲོག་མི་བརྟན་གཡོ། དོ་ནུབ་ཙམ་ཡང་མི་འཆིའི་གདེང་མེད་པས༑ ༑དེ་རིང་ཉིད་ནས་བརྩོན་པས་ཆོས་མཛོད་ཅིག ། སྲིད་པ་གང་དུ་སྐྱེས་ཀྱང་སྡུག་བསྔལ་སྣོད༑ ༑བདེ་བའི་སྐབས་མེད་མེ་འོབས་སྲིན་གླིང་འདྲ། འདི་ལས་ཐར་ཐབས་ད་རེས་མ་བརྩོན་ན། ཕྱིས་ནས་འགྱོད་པའི་དུས་ཤིག་འབྱུང་ཏ་རེ། སྲིད་པར་འདུ་བྱེད་ལས་ནི་དཀར་ནག་གཉིས། དེ་ཡི་རྩ་བ་བདག་ཏུ་འཛིན་པ་སྟེ། བདག་འཛིན་ཉོན་མོངས་གཉེན་པོས་མ་བཅོམ་ན། སྲིད་པའི་རྦ་ལས་ནམ་ཞིག་རྒལ་བར་འགྱུར།

མཆོད་གནས་སྡེ་གཉིས་གང་དུ་འདུག་ཀྱང་སྐྱོངས། མ་རྨས་གླིང་དཀར་རིགས་ཀྱི་མྱུ་གུ་ལས། ཉེར་འཁྲུངས་ས་ཡི་ཚངས་ཆེན་ཡོང་འདུའི་ལྗོན། སྡེ་བཞིའི་ཡལ་འདབ་རྒྱས་པས་མངའ་རིས་ཁྱོན། དགེ་བཅུའི་གྲིབ་བསིལ་སྣུམ་པོས་སྲུང་བར་མཛོད། དྲང་པོའི་མཁར་གནས་རྣམ་དཔྱོད་བསིལ་ཟེར་གྱིས། ཆབ་འབངས་ཕན་བདེའི་ཆུ་གཏེར་འཕེལ་བརྩོན་པའི། ཡ་རབས་མཛངས་བློན་དྲ་མའི་ཟླ་གཞོན་ཚོགས། ཕྲིན་ལས་སྣང་བ་གསར་པས་མཛེས་པར་མཛོད། དགེ་བཅུའི་བཀའ་ཁྲིམས་བཙན་པོའི་དུནྡྷུ་བྷིས། བག་མེད་གདོལ་སྤྱོད་གཉིད་ལས་ཆེར་བཟློག་པའི། དགེ་མཚན་སྣོད་བཅུད་དཔལ་དུ་ཉེར་ཤར་བས། རྫོགས་ལྡན་དུས་སྟོན་གསར་དུ་འཆར་བར་མཛོད། ཁྱད་པར་ཁྱོད་ལ་བརྟེན་ནས་བག་ཕེབས་སུ། སྤྲོ་སེམས་མགྲིན་པ་བཏེགས་ནས་རྒྱུུ་བ་ཡི༑ ༑ཉམ་ཆུང་རྩྭ་ཆུས་འཚོ་བའི་རི་སྐྱེས་ཚོགས། མི་འཇིགས་སྲོག་གི་སྦྱིན་པ་ཁྱོད་ཀྱིས་བྱིན༑ ༑ཡུལ་ཕྱོགས་སྤྱི་དང་བྱེ་བྲག་ལྗོངས་འདི་རུ། ནད་མུག་མཚོན་འཁྲུགས་བཙའ་སད་སེར་གསུམ་དང་། མི་རུང་ཐན་ལྟས་འབྱུང་བཞིས་འཇིགས་པ་སོགས། འགལ་བྱེད་གདུག་པའི་མིང་ཡང་མི་གྲགས་མཛོད། སྣོད་ལ་གཡང་ཆགས་ལོ་ཕྱུགས་རྟག་ཏུ་ལེགས། བཅུད་ལ་དཔལ་སྐྱེད་རྒབ་འཕེལ་བཞོན་བཅུད་རྒྱས། ཀུན་ཀྱང་བྱམས་པས་རྒྱུན་བརླན་དགེ་བཅུར་སྤྱོད། བསྐལ་པ་རྫོགས་ལྡན་དཔལ་ལ་ལོངས་སྤྱོད་ཤོག ། དེ་ལྟར་ཟག་མེད་བསང་གི་དགའ་སྟོན་གྱིས། དཀར་ཕྱོགས་ཟོ་དོར་གཉན་རྣམས་མཉེས་ཚིམས་ནས། རང་རང་སོ་སོའི་གནས་སུ་གཤེགས་ནས་ཀྱང་། སླད་ནས་མཆོད་སྦྱིན་མགྲོན་ལ་གཤེགས་པར་འཚལ། བཀྲ་ཤིས་ཀུན་གྱི་འབྱུང་གནས་མཆོག་གསུམ་ལྷའི། བཀྲ་ཤིས་བདེན་གསུང་ཆར་གྱི་རྒྱུན་བཟང་པོས། བཀྲ་ཤིས་སྲིད་ཞིའི་ཁོངས་སུ་རྟག་ཁྱབ་པས། བཀྲ་ཤིས་ཕན་བདེའི་རྩི་སྨན་རྒྱས་གྱུར་ཅིག ། ཅེས་སོགས་ཤིས་པ་བརྗོད་པས་དགེ་ལེགས་སུ་བྱའོ།

ས་སྐྱོང་དེ་ཡིས་དབུགས་དབྱུང་ཆེར་མཛད་ཀྱང་། རིང་པོར་མ་ཐོགས་དུས་ཀྱིས་གནམ་དུ་གཤེགས། རང་ལོ་དགུ་པ་སར་མ་ཐག་ཉིད་ནས། ལམ་དུ་ཞུགས་སྟེ་རིམ་གྱིས་དབུས་ཕྱོགས་བགྲོད། ལོ་དེ་ཆུ་ཕོ་སྤྲེའུའི་ལོ་ཡིན་སྟེ། འཕགས་པའི་ཡུལ་ན་ཨངྒི་ར་ཞེས་ཟེར། དེ་ཡི་དགུན་ར་མགོ་ཟླའི་མར་གྱིས་ངོའི། གྲལ་ཚེས་དགུ་པའི་གདུགས་ཕྱེད་ཉེ་བའི་ཚེ། དད་དགེ་དང་བཅས་བདུན་གྱི་ཚོགས་དབུས་སུ། དགེ་བསྙེན་བར་མ་རབ་བྱུང་སྔོན་འགྲོ་བས། དགེ་ཚུལ་ཡོངས་སུ་རྫོགས་པའི་བསླབ་པ་ནོས། གནས་ནི་ལྷ་བརྒྱད་གངས་ཀྱི་རྭ་བའི་དབུས། འབྲང་རྒྱས་རི་ཞོལ་ཡངས་ལྡན་གཙུག་ལག་ཁང་། མཁན་པོ་དོན་གྱི་སླད་ཏུ་མཚན་སྨོས་པ། རྒྱལ་བ་ཞྭ་དམར་དབང་གི་ཅོད་པན་ཅན། ཆོས་ཀྱི་གྲགས་པ་ཡེ་ཤེས་དཔལ་བཟང་ཉིད། དགེ་ཚུལ་ཉིད་ཀྱི་ལས་ཀྱི་སློབ་དཔོན་ནི། སྣར་ཐང་གདན་ས་སྡོམ་བརྩོན་ཤེས་དཔལ་ཡིན། ཆོས་རྒྱལ་དོན་གྲུབ་ཅེས་པར་མིང་དུ་བཏགས། མཁན་པོའི་དྲུང་དུ་སྤྱན་སྔ་ཕྱག་ཆེན་དང་། ལྔ་ལྡན་དྭགས་པོའི་ཆོས་བཞིའི་རྣམ་བཤད་ཐོབ། དགྱེས་པར་མཛད་ཅིང་ཟང་ཟིང་མང་དུ་གནང་། ཟླ་ཕྱེད་ཉེ་བའི་བར་དུ་བསྟེན་ནས་ཀྱང་། སླར་ཡང་དབུ་རུ་ལྷ་ལྡན་ཡུལ་གྲུར་ཕྱིན། དགུན་འབྲིང་རྒྱལ་གྱི་ཟླ་བའི་ཚེས་ལྔ་ལ། ཐུབ་ཆེན་ཆོས་ཀྱི་འཁོར་ལོའི་སྡེ་ཆེན་པོར། རྒྱལ་བ་ཀརྨའི་ཕྲིན་ལས་འཛིན་པའི་ཞབས། དགུང་ལོ་ང་བདུན་བཞེས་པ་ཅིག་མཇལ་སྟེ། ཐོག་མར་ཚེ་དཔག་མེད་མགོན་ལྷ་དགུ་མ། ཛེ་ཏ་རི་ཡི་ལུགས་ཀྱི་དབང་བསྐུར་མཛད། གྲུབ་རྒྱལ་མ་དང་ཏེ་བུའི་རྒྱུན་ཁྱེར་གནང་། འཇམ་དབྱངས་དཀར་ནག་མི་གཡོ་དཀར་པོ་དང་། འབྱུང་འདུལ་རྟ་མགྲིན་ཟླ་བ་རྒྱལ་མཚན་ལུགས། ཤྭ་ན་བཞི་བསྐོར་ཕག་མོའི་དབང་བཀའ་དང་། བེར་ནག་ལྕམ་དྲལ་ཆོས་སྐྱོང་ཕྲན་ལྔར་བཅས། ཚོགས་བདག་དམར་ཆེན་ཀུ་རུ་ཀུལླེ་དང་།

དེ་ཕྱིར་རྩ་བཞི་ཆང་ལྔ་སྲོག་ལྟར་སྲུངས། དུས་གསུམ་གཤེགས་པའི་རྒྱལ་རྣམས་འབྱུང་བའི་གནས། རྒྱལ་སྲས་ཀླུ་མཆོག་རྣམས་ཀྱིས་ཡོངས་བཟུང་བ། སྲིད་ཞིའི་འཇིགས་པའི་མེ་ཚོགས་གཞིལ་བའི་ཆབ། ཐེག་ཆེན་བྱང་ཆུབ་སེམས་མཆོག་བསྐྱེད་པར་མཛོད། འགྲོ་བ་མཐའ་དག་ཕ་མའི་རང་བཞིན་ལ། དགྲ་གཉེན་ཆགས་སྡང་འཁྲུལ་བའི་བློ་བཏང་སྟེ། བྱམས་དང་སྙིང་རྗེ་དགའ་བ་ཚད་མེད་པས། ཀུན་རྫོབ་བྱང་སེམས་བློ་སྦྱོང་སྙིང་པོར་མཛོད། རང་སྲོག་རང་ལ་ཕངས་པ་ཇི་བཞིན་ལ། གཞན་གྱི་སྲོག་ཀྱང་དེ་བས་གཅེས་ངེས་པས། རྣམ་གསུམ་དག་པའི་ཤ་ཡིས་ཚིམས་བྱོས་ལ། སྲོག་ལ་བབས་ཀྱང་གཞན་གནོད་བསམ་སྦྱོར་སྤོངས། སྨིན་བྱེད་དབང་བཞིའི་ས་བོན་བཏབ་པ་ལ། གྲོལ་བྱེད་རིམ་གཉིས་འབྲས་བཟང་གཡུར་ཟ་བར། དམ་ཚིག་རྣམ་དག་བརླན་དྲོད་གཙོ་ཆེ་བས། སྐུ་གསུང་ཐུགས་ཀྱི་དམ་ཚིག་ཉམས་མེད་མཛོད། གངས་ལྗོངས་གྲུབ་མཐའི་ཆུ་ཀླུང་ཇི་སྙེད་པ༑ ༑ཐུབ་བསྟན་ངེས་དོན་རྒྱ་མཚོར་རོ་གཅིག་ཕྱིར། སྡང་ཞེན་འཁྲུལ་པས་ཆོས་སྤོང་མི་གསོག་པར། དད་དང་དག་སྣང་ནོར་གྱིས་ཕྱུག་པར་མཛོད། ཐོས་པ་མ་རིག་མུན་སེལ་ཉིན་མོར་བྱེད། རྐུན་པོས་མི་འཕྲོག་རིན་ཆེན་གཏེར་གྱི་མཆོག ། འདོད་དགུ་འབྱུང་བའི་བུམ་བཟང་དཔག་བསམ་ཤིང་། ཟླ་བྲལ་ཡིད་བཞིན་ནོར་དེ་སྤྱི་བོས་ལོངས། དྲི་མེད་བསམ་བྱུང་ལྷ་གོས་པཉྩ་ལིས། ཐོས་མཆོག་བཻཌཱུར་དབང་པོ་ལེགས་ཕྱིས་ཏེ། འཁྲུལ་བའི་གཡའ་བྲལ་འཕྲོག་མིན་གཟི་འབར་བས། ཀུན་རྨོངས་མུན་པ་མ་ལུས་མིང་མཐར་བྱོས། དབེན་པ་གསུམ་གྱི་གོ་ཆ་ཆེར་བགོས་ནས། སོར་རྟོགས་ཡེ་ཤེས་རང་རིག་མདའ་རྣོན་པོས། བདག་འཛིན་ཉོན་མོངས་དགྲ་ཀུན་ཕམ་བྱས་ཏེ། གདོད་མའི་རྒྱལ་ས་ཚེ་འདིར་སྙོགས་པར་མཛོད། དེ་སྐད་བསླབ་བྱ་ཨུཏྤལ་གསར་པའི་ཕྲེང་། དོན་གཉེར་བློ་གསལ་གཞོན་ནུའི་མགུལ་རྒྱན་དུ༑ ༑པདྨ་བི་ཛའི་བློ་ཡི་མཚོ་ལས་ཕྱུང་།

འབྱུང་ཆེན་ལྔའི་རང་སར་ཐམས་ཅད་བསྡུད། ནམ་མཁའ་ཀ་ཀྴ་ཧ་ཨ་ཨཾ་རྣམས་མགྲིན། མེ་ནི་ཅ་ར་རི་ཥྚ ་ཨེ་རྣམས་རྐན། ས་ནི་ཏ་ལ་ས་ལི་ཨོ་རྣམས་ལྕེ། ཆུ་ནི་པ་ཟ་འ་ཨ༔ཨུ་རྣམས་མཆུ། རླུང་ནི་ཙ་ཡ་ཤ་ཞ་ཨི་རྣམས་སྣ། རྩ་བའི་སྡེ་ཚན་ལྔ་པོ་རང་སོར་བཞག ། གསུམ་དང་དབྱངས་ཡིག་བགོས་པས་འབྱུང་ཆེན་ལྔ། ཆུ་ཡི་ཆུ་དང་ས་ཡི་ས་འཐབ་ཕཊ། དྲག་པོ་དགྲ་གཤེད་འཕྲད་པས་དྲག་པོར་འགྱུར། ས་ཡི་ས་དང་ནམ་མཁའི་ནང་འཕྲད་ནི། མ་བུ་མཇལ་སྡེབ་བྱས་པས་ཞི་བར་འགྱུར། ནམ་མཁའི་སྡེ་ལ་ནམ་མཁའི་ནམ་མཁའ་བཞགས། རང་སྡེབ་བུ་ཡིས་བསྒྱུར་བས་དབང་དུ་བསྡུད། ཆུ་ཡི་སྡེ་ལ་ས་ཡི་ས་དག་སྣུན། རང་གཤེད་རང་གི་བསྐྱེད་པས་རྒྱས་པར་འགྲོ། དེ་བཞིན་གསང་སྔགས་ཀུན་ལ་ཤེས་པར་བྱ། ༈ ཡི་གེ་འཁོར་ལོ་རྫོགས་པའི་དོན་བཤད་པ། རྒྱལ་བའི་དཀྱིལ་འཁོར་ས་བཅུ་འདས་པའི་གོང་། གསང་སྔགས་རྒྱལ་པོ་མཆོག་གི་ཕོ་བྲང་ལ། བསྭོ་དང་ཨཱོྃ་གཉིས་རྫོགས་པར་ཡོངས་སུ་བཞུགས། འདབ་བཞི་སྭ་ཧཱ་ནི་རི་བཞི་ཡོད་དེ། འདབ་བརྒྱད་པུས་པ་ལི་དང་དྷ་བ་གནས། ཕྱི་འཁྱམས་བར་འཁྱམས་ཀུན་ལ་དབུས་སུ་ཨོཾ། ཕྱོགས་ལ་སྭ་ཧཱ་ནི་རི་གནས་པའོ། ཟུར་བཞི་ཐུགས་རྗེ་རྒྱུན་གྱི་ཕོ་བྲང་ལ། དྷཀི་དབུས་ལ་མཚམས་བཞི་ལ་ཡིན་ནོ། ཕྱོགས་བཞི་ཕོ་བྲང་བཞི་ལ་ཧཱུཾ་བཞི་གནས། དེ་བཞིན་མ་ཡང་བཞི་རུ་ཤེས་པར་བྱ། དེ་དག་གཞི་དང་འབྲས་བུའི་ས་བོན་ཡིན། ལམ་གྱི་འཕྲུལ་ཆེན་བཞི་བཅུ་ཡས་བགྲང་པ། ཀྐྷཱ་ཁྑྷཱ་གྒྷཱ་ངྔྷཱ་བཞི་ནི་སྒོ་བཞིར་གནས། ཅྕྷཱ་ཆྖྷཱ་ཇྗྷའ་ཉྙྷཱ་བཞི་ནི་ཕྱོགས་བཞི་སྟེ། ཏྟྷཱ་ཐྠྷཱ་དྡྷཱ་ནྣྷཱ་དུར་ཁྲོད་མཚམས་བཞིའོ། པྤྷཱ་ཕྥྷཱ་བྦྷཱ་མྨྷཱ་སྨྲ་བར་འཁྱམས་སྒོ་མ་བཞི། ཙྩྷཱ་ནི་ཤར་ལ་ཚྪྷཱ་ནི་བྱང་ཡིན་ཏེ། ཛྫྷཱ་ནི་ནུབ་ལ་ཝྺྷཱ་ནི་ལྷོ་རུ་བཀོད། ལྷོ་ཤར་བྱང་ཤར་ཞྮྷཱ་ལ་བྱང་ནུབ་ལྷོ་ནུབ་དུ།

བྱང་ཆུབ་ཀྱི་གོ་འཕང་གི་རྒྱུ་རྒྱལ་བའི་སྲས་ཀྱི་བསླབ་བྱ་ཕྃ་རོལ་ཏུ་ཕྱིནྃ་པ་དྲུགྃ་གིྃ་སྤྱོད་པ་ལ་འཇུག་པས་མི་མཐུན་ཕྱོགས་སྤང་བ་དང་། མཐུན་ཕྱོགས་དང་དུ་ལེན་མི་ཤེས་པའི་བླང་དོར་ལྟ་བའི་འྃདྲེན་བྱེྃད་དེ་མིག་ཅེྃ་རེྃ་ལོངྃས་པས་ལམ་ལོག་པར་འགྲོ་བ་མ་ཚོར་བའི་ངང་། ལུས་ཀྱི་ཕྱི་ནང་གི་འབྱུངྃ་བྃ་སྡུད་པའི་ཐིམྃ་རིམྃ་གྱི་སྣང་བ་དང་བཅས་འཇིག་རྟེན་ཕྱི་མའི་ལམ་པོ་ཆེར་ཞུགས་དགོས་པའི་དུས་བྱུངྃ་ནྃ་རང་སྟོབས་ཀྱི་དགེ་རྩ་ནི་མེད་གཞན་སྟོབས་ཀྱི་སྐྱབས་གནས་དང་། བྲལ་བས་སུྃ་ལྃ་ཡང་རེྃ་ས་དང་དཔུང་གཉེན་བྱེད་པ་མ་མཆིས་པས་སྐྱབས་གནས་གང་ལ་བྱེད་མི་ཤེས་པས་ཉམ་ཐག་པ་ལ་ཐུགྃས་རྗེྃས་གཟིྃགས་ཤིྃྃག་སོགས་འགྲེ། དྲུག་པ་དགེ་བ་སྟོབས་ཆུང་བ་ལ་བསམ་ནས་ཆོས་ལ་བསྐུལ་བ་ནི། རྣམ་དཀར་གྱི་བག་ཆགས་ཅུང་ཟད་དང་ལྡན་པ་འགའ་ཞིག་སྒོ་གསུམ་དགེྃ་བ་བྃཅུ་ཡི་ལམྃ་སྒྲུབ་པ་ལྃ་ཕྱོགས་པའི་མོསྃ་པསྃ་ཆོས་སྒོར་ཞུྃགས་པ་ལྟར་བྱས་ནྃ་ཡངྃ། བསམ་སྦྱོར་ཐམས་ཅད་རྣམ་པར་དག་མ་ནུས་པ་གཟུགས་བརྙན་ཙམ་དུ་གྱུར་ཏེ། ཁ་ནང་དུ་ཕྱོགས་ཏེ་ཀུན་སློང་ལ་བརྟགས་པས་རྣམྃ་པརྃ་དཔྱྃད་ནྃ་འགྲན་སེམས་ཕྲག་དོག ། རྙེད་བཀུར་གྲགས་འདོད་སོགས་ཀྱིས་ལྟར་སྣང་འཇིག་རྟེན་ཆོསྃ་བྃརྒྱད་བསྒྲུབ་ཕྱིར་ཙམ་གྱི་ཀུན་སློང་དུག་ཅན་དང་། སེམས་བསྐྱེད་ཀྱིས་རྩིས་མ་ཟིན་པ་མགོ་བསྐོར་བའི་ཟོབྃ་དངྃ་འདྲེྃས་པར་གྱུར་ཅིང་སྡིག་པ་མིྃ་དགེྃ་བ་ལ་ནི་ཆེད་དུ་བརྩོན་མི་དགོས་པར་ངམ་ངམ་ཤུགས་ཀྱིས་དེ་ལ་ཞུགས་པས་རྣམྃ་སྨིྃན་ནི་སྡིག་པའི་ལས་ཕྲ་མོས་ཀྱང་སྡུག་བསྔལ་མི་བཟད་པ་ཡོད་པ་མྃ་ཚོརྃ་བའམ་མ་ཤེས་པས་ཉེས་སྡིག་གིས་ཤུགྃས་ཀྱིྃས་འགོྃ་བར་གྱུར། དེའི་ངང་ནས་ཡུད་ཙམ་གྱིས་འཇིག་རྟེན་ཕྱི་མར་སོན་ཏེ། སྐྱེ་ཤི་གཉིས་ཀྱི་སོ་མཚམས་བརྃ་དོྃར་སོན་ནས་འཁྲུལ་སྣང་དང་སྡུག་བསྔལ་འབའ་ཞིག་གིས་སྐད་ཅིག་ཀྱང་སོ་མི་བསོད་པ་བྱ་སྤུ་རླུང་གིས་ཁྱེར་བ་བཞིན་འཇིགས་སྐྲག་གིས་ལུས་སེམས་དམྱལྃ་ཐྃག་གཅོྃད་པའམ་མནར་བའི་དུྃས་འགྱོད་པས་སླེབས་ཀྱང་ཆོས་ལ་བྱ་ལོང་མེད་པས་སྣང་བ་ཐམས་ཅད་དགྲར་ལངས་བྱུང་ན་དེ་ལས་སྒྲོལ་བའི་དེད་དཔོན་སུྃ་ལྃ་རེྃ་ལྟོས་འཆའ།

ཚོང་སོ་ནམ། འདོད་ཡོན་གྱིས་གཡེྃང་བའིྃ་ཟིན་མེད་ཆུ་ཡི་རི་མོ་ལྟ་བུ་ལ་དཀའ་ཚེགས་ཆེན་པོས་གོམས་བཞིན་དེ་ལ་སྲེད་པ་ཇེ་ཆེར་འཕེལ་ཏེ་རྟག་པར་དེའི་དོན་སྒྲུབ་འབའ་ཞིག་གི་ཁོལྃ་པོརྃ་འཕྱན་ཏེ་མིྃ་ཚེྃ་ཟྃད་པ་མ་ཚོར་བའི་ངང་ནས་དལ་འབྱོར་གྱི་མི་ལུས་འདི་སྟོང་ཟད་དུ་འགྲོ་བ་ལ་འཁོར་དང་ལོངས་སྤྱོད་མངའ་ཐང་སྙན་གྲགས་སྟོབས་མངའ་རིས་སོགས་ཐབས་གང་གིས་བཟློག་ཏུ་མེད་ཅིང་། དགྲྃ་བོ་ཆེྃན་པོ་འཆིྃ་བདག་དེ་གློ་བུར་དུ་ཐོག་མགོ་ལ་བབས་པྃ་བཞིན་བྱུངྃ་ནྃ་འཇིག་རྟེན་འདིར་སྣང་གཞན་ལྟ་ཅི་མི་ཚེ་རིལ་པོ་གཅེས་པར་བསྐྱངས་ཤིང་མེ་སྟག་ཕྲ་མོ་ཕོག་ཀྱང་བཟོད་དུ་མི་ཐུབ་པའི་ལུས་ཀྱང་ཤུལ་དུ་བོར་ནས་རྣམ་ཤེས་གཅིག་པུ་མར་གྱི་དཀྱིལ་ནས་སྤུ་བཏོན་པ་བཞིན་ཆ་མེད་ཀྱི་ཡུལ་དུ་རྒྱུས་མེད་གཅེར་བུ་ལག་སྟོང་མཆན་དུ་བཅུག་པ་ལས་མེད་པའི་དུས་ཞིག་བྱུང་ན་རེ་ལྟོས་ཞུ་ཡུལ་སུྃ་ལྃ་རེྃ། འཇིགས་སྐྲག་སྡུག་བསྔལ་ཆེན་པོ་དེ་འདྲ་བ་འབྱུང་བའི་སྐབས་སུ་དེ་ལས་སྐྱོབ་པའི་མྱུར་མགྱོགས་ཀྱི་ཐུགྃས་རྗེྃས་གཟིྃགས་ཤིྃག་སོགས་འགྲེ། ལྔ་པ་ལས་རྒྱུ་འབྲས་ལ་བསམ་པའི་བྱེ་བྲག་སྡིག་པ་སྟོབས་ཆེ་བ་ལ་བསམ་པས་ཆོས་ལ་བསྐུལ་བ་ནི། ཁྱེད་ཀྱི་གདུལ་བྱ་བོད་ཡུལ་དུ་གནས་པ་ཕལ་ཆེར་བྱིྃས་པྃ་སོ་སོའི་སྐྱེ་བོ་བསམ་མནོ་མ་ཟིན་པའི་རངྃ་བྃཞིན་གྱི་སྒོ་གསུམ་གྱི་བྱ་སྤྱོད་མ་བརྟགས་བྱུང་རྒྱལ་དུ་སྨྱོན་པ་ལྟར་བྱེད་པའི་སོྃ་སོྃའི་སྐྱེྃ་བོྃ་ཤ་སྟག་སྟེ། དེ་དག་ཀུྃན་གྱི་རང་གཤིས་ནི་ཐར་ལམ་གྱི་རྒྱུ་བླང་དོར་མིག་འབྱེད་པའི་ཐོསྃ་པ། ངེས་པའི་དོན་གཏན་ལ་འབེབས་པའི་བསྃམ་པ། ཟབ་དོན་ཉམས་སུ་ལེན་པ་སྒོམྃ་པའིྃ་ཆོས་ཀྱི་བྱ་སྤྱོད་ལ་མོས་བརྩོན་ཆུང་བས་དམ་པའི་ཡོན་ཏན་གྱི་ནྃ་ཚོྃད་ནི་ཆེསྃ་ཆེརྃ་རྒུྃད་པའི་གོམས་འགྲོས་འཁྱོར་ཞིང་ཐར་པའི་གནས་སུ་བགྲོད་པར་མི་ནུས།

བདག་ལ་བདེ་ཆེན་སྦྱོར་བཞེད་མཛོད། ཁྱོད་མེད་ན་ནི་བདག་འགུམ་པས། ཀྱེ་ཡི་རྡོ་རྗེ་ཁྱོད་བཞེངས་ཤིག ། སྟོང་པའི་རང་བཞིན་ཉིད་སྤངས་ལ། རི་ཁྲོད་མ་འབྲས་གྲུབ་པར་མཛོད། དགའ་གཙོ་འཇིག་རྟེན་མགྲོན་གཉེར་བ། སྟོང་པ་ཉིད་ལ་བཞུགས་སམ་ཅི། ཁྱོད་མེད་ཕྱོགས་ནི་མི་འཚལ་བས། གདོལ་པ་མོ་བདག་ཞུ་བ་བགྱིད། བདག་གིས་ཁྱོད་ཀྱི་ཐུགས་འཚལ་གྱིས། མིག་འཕྲུལ་མངའ་བ་ཁྱོད་བཞེངས་ཤིག ། གཡུང་མོ་བདག་ཉིད་དྲན་ཉམས་པས། སྙིང་རྗེ་རྒྱུན་ཆད་མ་མཛད་ཅིག༈ ། གླང་པོ་རྟ་བོང་གླང་རྔ་མོ། མི་དང་ཤ་ར་བྷ་བྱི་ལ། གཡས་པའི་ཐོད་པ་བརྒྱད་པོ་ལ། གཉིས་འཐུང་ལ་སོགས་རིམ་ཤེས་བྱ། ས་དང་ཆུ་དང་རླུང་དང་ནི། མེ་དང་ཟླ་བ་ཉི་མ་ཉིད། མཐར་བྱེད་དང་ནི་ནོར་སྦྱིན་ཉིད། གཡོན་པའི་ཐོད་པ་བརྒྱད་ནའོ། སྒེག་ཅིང་དཔའ་བོ་མི་སྡུག་པ། དགོད་ཅིང་དྲག་ཤུལ་འཇིགས་རུང་བ། སྙིང་རྗེ་རྔམ་དང་ཞི་བ་ཡིས། གར་དགུའི་རོ་དང་ལྡན་པ་ཉིད། ཨྂ་དང་ཧཱུྃ་གིས་རྡོ་རྗེ་ཆེ། ཁུ་བའི་གཟུགས་ལས་བཞེངས་ནས་ནི། ཞབས་རྣམས་ས་ལ་རྡེབས་པ་དང་། ལྷ་དང་ལྷ་མིན་བསྡིགས་པར་མཛད། གྂ་ཙྂ་བྂ་གྷྂ་པྂ་ཤྂ་ལྂ་ཌྂ། འདི་ཡིས་ས་བོན་ཕྱུང་བ་དང་། བདག་པོའི་ས་བོན་དག་གིས་ནི། ཨྂ་ཧཱུྃ་སྔོན་པོ་འབར་འཇིགས་པས། མ་མོའི་འཁོར་ལོ་གྲོངས་ཉམས་དགར། ཁ་དོག་ནག་པོ་འཇིགས་ཆེན་པོ། བདག་མེད་བདེ་བ་སྦྱིན་པ་པོ། གཙོ་བོ་འདི་ལྟར་བསྒོམ་པ་ཉིད། དཀར་མོ་གཡས་ན་གྲི་གུག་སྟེ། དེ་བཞིན་གཡོན་ན་རོ་ཧི་ཏ། ཆོམ་རྐུན་མ་གཡས་ཅང་ཏེའུ་སྟེ། གཡོན་པའི་ཕྱག་ན་ཕག་པ་ཉིད། རོ་ལངས་མ་གཡས་རུས་སྦལ་ཏེ། གཡོན་པ་ཡིས་ནི་པདྨའི་སྣོད། གྷསྨ་རཱི་ཡི་གཡས་ན་སྦྲུལ། གཡོན་པས་རྣལ་འབྱོར་ལྷུང་བཟེད་ཉིད། པུཀྐ་སཱི་གཡས་སེང་གེ་སྟེ། གཡོན་པས་དགྲ་སྟྭ་དེ་བཞིན་ནོ། རི་ཁྲོད་མ་གཡས་དགེ་སློང་སྟེ།

འཐབ་པ་གཅིག་པུའི་ལུགས་ཀྱང་གཞན་ལས་བཤད། དེ་སོགས་རྣམ་བཞག་མཐའ་ཡས་འདིར་མ་བརྗོད། ཞིབ་པར་གཞུང་དང་བླ་མའི་མན་ངག་དང་། རང་གི་རིག་པས་ལེགས་པར་དཔྱོད་ཅིག་གུ། ཨེ་མ་དོན་ལ་མེད་བཞིན་སྣ་ཚོགས་སྣང་། བྱེད་པོས་དབེན་ཀྱང་བདེ་སྡུག་ལས་ཀྱི་མཐུ༑ ༑རྟག་དངོས་ཁེགས་ཀྱང་རྒྱུན་གྱི་ཆད་མཐའ་སངས། ཐ་སྙད་ཚུལ་འདི་དོན་དམ་འཇུག་སྒོ་རྨད། ཕྱི་ནང་ཆོས་ཀུན་རྟེན་ཅིང་འབྲེལ་འབྱུང་ཚུལ། རྣམ་ཀུན་མི་བསླུ་ཕན་གནོད་ཚད་མའི་འབྲས། མཐོང་ཆོས་ཞིང་སར་སྨིན་འདི་མ་བཟོད་ནས། མགྲིན་སྔོན་དཀའ་ཐུབ་ཟོལ་གྱིས་ངལ་བ་བརྟེན། རབ་དྭངས་གཙུག་ལག་རྩིས་གཞུང་ཨ་ད་ཤར། ལྐོག་གྱུར་གཟུགས་བརྙན་འབུམ་ཕྲག་འཆར་བའི་དཔལ། མཐོང་བས་མ་ངེས་ལེགས་བྲིས་གཙོ་བོ་དེས༑ ༑རྣམ་བཀྲའི་འདྲེན་བྱེད་སྟོང་དུ་འཕྲུལ་བར་ངེས། དེ་སླད་ཉེར་མཁོ་རྩིས་ཀྱི་རེའུ་མིག་ཕྲེང་། གཞུང་བཟང་ཟླ་བའི་ཟེར་གྱིས་ལེགས་ཕྱེ་བའི། ཉམས་མཚར་ཀུ་མུ་ད་༼དཱ༽ཡི་རྫིང་བུ་འདི། དོན་མཐུན་སྐལ་བཟང་གྲོགས་ལ་བརྩེ་བས་འབུལ། ལེགས་བྱས་ར་མ་ཎ་ཡི་ཉག་ཕྲན་འདིས། ལོག་སྤྱོད་ཉེས་བརྒྱའི་ལངྐའི་གྲོང་ཁྱེར་འབུམ། རབ་བཅོམ་རྫོགས་ལྡན་གྲགས་དཀར་དབྱར་གྱི་རྔས། ས་གསུམ་མདོངས་མཐའ་དགའ་བའི་གར་བྱེད་ཤོག ། མཆོག་གསུམ་བདེན་མཐུ་འགྲོ་བའི་རྒྱུད་ལ་ཞུགས། ནག་ཕྱོགས་སྟོབས་ཞི་དཀར་ཕྱོགས་དབུ་རྨོག་བཙན། རྒྱུ་རྐྱེན་རྟེན་འབྲེལ་དཔྱིད་བཏབ་ས་བོན་བཞིན། དོན་གཉིས་གསེར་གྱི་སྙེ་མ་གཡུར་སྨིན་ཤོག ། བཀྲ་ཤིས་ཀུན་གྱི་བཀྲ་ཤིས་དཔལ་གྱུར་བ། གྲགས་པའི་བཀྲ་ཤིས་སྙན་གྲགས་ཆེན་པོ་མཆོག ། འཇམ་དཔལ་ཡེ་ཤེས་སེམས་དཔའི་བཀྲ་ཤིས་དེས། ཕྱོགས་དུས་ཀུན་ན་བདེ་ལེགས་འཕེལ་གྱུར་ཅིག ། ཅེས་པའང་ཉེར་མཁོའི་ཡན་ལག་དང་ཉམས་མཚར་ཙམ་དུ། ཟེ་ཆེན་པ་འཇམ་དབྱངས་བློ་གྲོས་རྒྱ་མཚོས་འཕྲལ་མར་བྲིས་པ་དགེ་ལེགས་ཀྱི་སྣང་བ་ཆེན་པོས་ཕྱོགས་ཀུན་ཏུ་ཁྱབ་པའི་རྒྱུར་གྱུར་ཅིག སརྦདཱ་མདྒལཾ་བྷ་ཝནྟུ།

རྗེས་ཀྱི་གཏང་རག་གི་བསྟོད་པའི་སྐབས། བརྗོད་མེད་ཤེས་རབ་མི་གཡོ་ཆོས་ཀྱི་སྐུ༔ བདེ་ཆེན་ལོངས་སྤྱོད་རྫོགས་སྐུ་རིགས་ལྔའི་གཙོ༔ ཐུགས་རྗེ་ཐབས་མཁས་རྒྱ་ཆེ་སྤྲུལ་པའི་སྐུ༔ ཞི་ཁྲོ་སྐུ་གསུམ་ལྷ་ལ་ཕྱག་འཚལ་བསྟོད༔ ཅེས་ཤྭ་ལོ་ཀ་གཅིག་པུས་བསྟོད། རྟེན་ཡོད་མེད་དང་འབྲེལ་བརྟན་བཞུགས་གཤེགས་གསོལ་གང་འོས་བྱ། བདག་བསྐྱེད་བསྡུ་ལྡང་དགེ་བསྔོ་ཤིས་བརྗོད་སོགས་གཞུང་ལྟར་བྱས་པས་གྲུབ་བོ། ཞི་ཁྲོ་རབ་འབྱམས་རྒྱལ་བའི་ལྷ། དཀྱིལ་འཁོར་གཅིག་ཏུ་བསྒྲུབས་བྱས་ཏེ། དམ་ཚིག་ཉམས་ཆག་གསོ་བའི་ཐབས། གཏེར་གཞུང་ཇི་བཞིན་བསྡེབས་པའི་དགེས། བདག་ཅག་སྔགས་ཀྱི་སྒོར་ཞུགས་རྣམས། རྩ་བ་ཡན་ལག་གིས་བསྡུས་པའི། ཉམས་ཆག་མ་ལུས་སོར་ཆུད་དེ། སྐུ་ལྔའི་འབྲས་བཟང་མྱུར་སྨིན་ཤོག ། ཅེས་པའང་རྩ་གསུམ་པདྨ་གསང་ཐིག་དང་། ཐུགས་སྒྲུབ་བདེ་གཤེགས་འདུས་པ་གང་ལའང་སྦྱར་དུ་རུང་བའི་ཞི་ཁྲོའི་སྐོང་བཤགས་ཀྱི་ཆོ་ག་འདི་ཡང་གཏེར་གཞུང་རྡོ་རྗེའི་གསུང་ལ་འཐད་ལྡན་མན་ངག་གིས་ཟུར་བརྒྱན་ཏེ་བདུད་འཇོམས་ནམ་མཁའི་རྡོ་རྗེས་ངག་ནས་སྨྲས་རིམ་བཞིན་བློ་གསལ་དཔལ་ལྡན་མཁས་མཆོག་ནས་དག་ཅིང་མགྱོགས་པར་བྲིས་པ་ཁམས་གསུམ་སེམས་ཅན་ཐམས་ཅད་ཀུན་བཟང་པདྨཱཀ་རའི་གོ་འཕང་ལ་འགོད་པའི་རྒྱུར་གྱུར་ཅིག ། སིདྡྷི་རསྟུ། རིགས་དྲུག་གནས་སྦྱོང་གི་སྲེག་གཏོར་སྦྱོང་བའི་དམིགས་བཟླས་ཟུར་བྱང་ནི། མདུན་བསྐྱེད་དཀྱིལ་འཁོར་གཙོ་དང་འཁོར། ཡབ་ཡུམ་སྙོམས་ཞུགས་བྱང་སེམས་སྤྲིན། འོད་ཟེར་མེ་ལྕེ་ལྟ་བུར་འཕྲོས། རང་གི་ལུས་ཀྱི་གནས་དྲུག་ཏུ། རིགས་དྲུག་ས་བོན་དྲུག་་་རྐང་མཐིལ་གཡས་གཡོན་དུ་དུ། གསང་བར་པྲེ། ལྟེ་བར་ཏྲི། སྙིང་གར་ནྲྀ། ཨོག་མར་སུ། དཔྲལ་བར་ཨ། ཐོག་མེད་དུས་ནས་བསགས་པ་ཡི། ལས་དང་བག་ཆགས་ཐམས་ཅད་བསྡུས།

རྩོད་ལྡན་དུས་མཐའི་མར་གཤམ་སྡུག་བསྔལ་གྱི། མེ་འོབས་ནང་དུ་ཆུད་འདྲའི་ཉམ་ཐག་འགྲོར། རྒྱལ་རྣམས་ཐུགས་རྗེའི་དཔྱང་ཐག་རིང་བའི་དུས། ཨེ་མ་སངས་རྒྱས་གཉིས་པ་དངོས་སྣང་བའི། མཆོག་གྱུར་གླིང་འདིའི་གདུང་སེལ་བསིལ་སྦྱིན་དབང་། ལས་སྨོན་སྐལ་པས་ཐོབ་འདི་སྟོང་གསུམ་ཁམས། རིན་ཆེན་ནོར་སྣས་བཀང་བྱིན་མཚོན་ནུས་མིན། གང་གི་རྣམ་ཐར་ཡོན་ཏན་མཛད་ཕྲིན་ཆ། རྒྱལ་རྣམས་བརྗོད་དཀའ་སོ་སྐྱེས་ལྟ་ཅི་སྨྲོས། འོན་ཀྱང་འཇམ་མགོན་མཁན་སློབ་རྣམ་གཉིས་ཀྱི༑ ། མཁྱེན་རབ་ཉི་ཟླའི་སྣང་བས་གསལ་མཛད་པའི། སྔོན་གནས་རྗེས་དྲན་ཡབ་སྲས་བླ་སློབ་འབྲེལ། ཐུགས་ཡིད་གཅིག་འདྲེས་ཟབ་གཏེར་སྒོ་བརྒྱ་གྲོལ། རྡོ་རྗེའི་ལུང་དོན་བསླུ་མེད་བདེན་པས་གྲུབ། རྣམ་ཐར་རྒྱ་མཚོའི་རྦ་རླབས་དགུང་དུ་འཁྱིལ། དེ་ལས་ཐུན་མོང་ཆུ་བུར་སྣང་ངོ་རུ། མཛད་ཕྲིན་ཕྱི་ཚུལ་མོས་བློས་གང་དཔོག་པ། དོན་བསྡུས་གོ་བདེའི་ཆེད་དུ་སྡེབས་ཤིག་པར། མཚམས་སྦྱོར་བགྱིས་ལ་ནོངས་གྱུར་༄༅། གང་མཆིས་བཤགས། དགེ་བས་མདོ་སྔགས་རྒྱལ་བསྟན་དར་ཞིང་རྒྱས། རྒྱལ་ཁམས་ཀུན་བདེ་དགེ་བཅུའི་ཁྲིམས་ལ་གནས། ཁྱད་པར་མཆོག་གཏེར་གདམས་པའི་རིང་ལུགས་ཀྱིས༑ ། འཛམ་གླིང་ཀུན་ཁྱབ་འབྲེལ་ཚད་དོན་ལྡན་ཤོག ། བདག་གཞན་འབྲེལ་ཐོག་མ་གྱུར་མཁའ་མཉམ་འགྲོ། ཚེ་འདིར་བདེ་སྐྱིད་ཆོས་མཐུན་བསམ་དོན་འགྲུབ། ཕྱི་མ་པདྨས་ཁེབས་པའི་ཞིང་སྐྱེས་ནས། པདྨའི་མྱུ་གུའི་མཛད་ཕྲིན་ཡོངས་འགྲུབ་ཤོག ། མཁའ་ལ་རྩ་གསུམ་བྱིན་རླབས་སྤྲིན་ཆེན་འཁྲིགས། ས་ལ་རྫོགས་ལྡན་ཕན་བདེའི་གྲུ་ཆར་འཇོ། བར་ལ་སྐྱེ་རྒུས་དགའ་སྐྱིད་གླུ་བྲོ་འཁྲབ། མཐར་ཐུག་འཁོར་མཐའ་སྟོང་པའི་བཀྲ་ཤིས་ཤོག ། ཅེས་པའང་གཏེར་ཆེན་ཆོས་ཀྱི་རྒྱལ་པོའི་མཛད་ཕྲིན་སྐོར་སོ་སྐྱེས་བློ་རྟོག་གིས་གཞལ་བའི་སྤོབས་པ་བྲལ་ཀྱང་།

བདག་ཅག་གངས་ཅན་བོད་དུ་སྐྱེས་ནས་དལ་འབྱོར་གྱི་ལུས་ཐོབ་ཆོས་ཀྱི་མཐུན་རྐྱེན་ཕུན་སུམ་ཚོགས་ཀྱང་རང་རྒྱུད་ལས་ངན་གྱིས་སྙིགས་དུས་སུ་སྐྱེས། མཐའ་འཁོབ་མུན་པའི་གླིང་འདྲེ་སྲིན་ནག་ཕྱོགས་ཀྱིས་གང་བ་བོད་ཡུལ་ལྟ་བུ་འདུལ་བའི་ཐུགས་བསྐྱེད་དང་། བསྐལ་པ་གྲངས་མེད་དུ་ཚོགས་གཉིས་རྫོགས་སྨིན་སྦྱངས་པས་བརྟན་པའི་སྨོན་ལམ་དུས་སུ་སྨིན་པས་སངས་རྒྱས་ལས་ཀྱང་བཀའ་དྲིན་ཆེ་བའི་ཐུགྃས་རྗེྃ་ལྡནྃ་པའིྃ་རྒྱལྃ་བྃ་ཁྱེྃད་ལྟྃ་བུྃ་ཡང་སྔོན་བོད་འདིར་སྐུ་དངོས་སུ་ཕེབས། རྫུ་འཕྲུལ་ཆེན་པོས་མི་དང་མི་མིན་བཏུལ། ལས་ཅན་གྱི་གདུལ་བྱར་སངས་རྒྱས་ལས་ཀྱང་དཀོན་པའི་གསང་སྔགས་ཀྱི་ཆོས་འཁོར་བསྐོར། ལུགས་གསུམ་མི་འགྱུར་ལྷུན་གྱིས་གྲུབ་པའི་གཙུག་ལག་ཁང་བཞེངས། ལོ་པཎ་བརྒྱ་རྩ་ཞལ་འཛོམས། སངས་རྒྱས་ཀྱི་བསྟན་པ་ཉི་མ་ཤར་བ་ལྟ་བུའི་དཀར་འཇམ་ཡུག་གཅིག་ཏུ་བརྡལ་བར་མཛད་ཀྱང་། སྐུ་དངོས་ཀྱི་གདུལ་སྐལ་རྫོགས་ནས་ཕྱི་མའི་བོྃད་འབངྃས་ཐམས་ཅད་བོརྃ་བ་ལྟར་མཛད་ནསྃ་ལྷོ་ནུབ་སྲིན་པོའི་ཁ་གནོན་ཕྱིར་རྔྃ་ཡབྃ་སྲིན་གླིངྃ་དུྃ་གཤེྃགས་ནས། ད་ལྟའི་དུས་སྐབས་འདིར་མཛད་པའི་ལྷག་མ་ཙམ་ལས་མཇལ་བའི་སྐལ་བ་མེད་སྟེང་། དུས་མཐའི་རང་རྟགས་ནད་མུག་མཚོན་འཁྲུགས་ཀྱིས་བསྟན་འགྲོའི་བདེ་སྐྱིད་སྣུམ་ཟད་མར་མེ་ལྟར་གྱུར་པས་བོད་ཁ་བ་ཅན་འཕགས་པ་སྤྱན་རས་གཟིགས་ཀྱི་ཞིང་ཁམས་སུ་གདོངྃ་དམྃར་བོྃད་ཕྲུྃག་མིའི་འགྲོ་བ་དང་པོར་མཆེད་པའི་ཚེ། འཕགས་པ་ཉིད་སྤྲེལ་སྒོམ་བྱང་ཆུབ་སེམས་དཔའ་དང་། སྒྲོལ་མའི་སྤྲུལ་པ་བྲག་སྲིན་མོ་དང་ཐབས་ཤེས་སུ་འབྲེལ་བ་ལས་བྱུང་བས་ཕ་སྤྲེལ་ཡིན་པའི་དབང་གིས་དྲན་རིག་རྣོ། ཁྲོ་གཏུམ་ཆེ། བློ་མི་བརྟན། མ་སྲིན་མོར་གྱུར་པས་ཤ་ཁྲག་ལ་དགའ་བའི་སྤྲེྃའུྃའི་རིགས་བརྒྱུད་རིམ་གྱིས་མི་རུ་གྱུར་པའི་ཚྃ་བོྃ་ལྟ་བུ་རྣམྃས།

དེ་ནས་སའི་འཁོར་ལོ་ལ་ཏའི་སྡེ་པ་ལྷག་པའི་ལྷ་སྟེ་དེའི་ཏྻཱ་རླུང་དུའོ། དབང་ལྡན་དུ་ཐཱུ་དང་བདེན་བྲལ་དུ་དྼཱྀ་དང་མེར་དྷཱི་དང་། ས་འོག་ཏུ་ནཱ་སྟེ་རླུང་གི་སྒོའི་དུར་ཁྲོད་ཀྱི་འཁོར་ལོའི་དབུས་སུའོ། ཞེས་པ་ནི་བུ་མོའི་དུས་སྦྱོར་ལའོ། དེ་བཞིན་དུ་སྲང་ལ་སྟེང་དུ་ན་སྟེ་ནུབ་ཀྱི་སྒོར་རོ། ཤར་དུ་དྷི་དང་ལྷོར་དྼི་དང་བྱང་དུ་ཐུ་དང་ནུབ་ཏུ་ཏླཱིའོ། དེ་ནས་ཆུའི་ཁམས་ཀྱི་འཁོར་ལོ་ལ་པའི་སྡེ་པ་ལྷག་པའི་ལྷ་སྟེ་དེའི་མྺྀ་རླུང་དུའོ། དབང་ལྡན་དུ་ཕཱུ་དང་བདེན་བྲལ་དུ་བྼཱྀ་དང་མེར་བྷཱི་དང་ས་འོག་ཏུ་མཱ་སྟེ་དབང་ལྡན་གྱི་སྒོ་རུའོ། དེ་ལྟར་ཀརྐ་ཊའི་དུས་སྦྱོར་ལ་ལྷག་པའི་ལྷ་རྣམས་སོ། དེ་བཞིན་དུ་སེང་གེའི་དུས་སྦྱོར་ལ་ནམ་མཁར་མ་སྟེ་བྱང་གི་སྒོ་རུའོ། ཤར་གྱི་རྩིབས་ལ་བྷི་དང་ལྷོར་བྼྀ་དང་བྱང་དུ་ཕུ་དང་ནུབ་ཏུ་བླྀའོ། དེ་ནས་མེའི་འཁོར་ལོ་ལ་ཊའི་སྡེ་པ། ལྷག་པའི་ལྷ་སྟེ་དེའི་ཊླཱྀ་རླུང་གི་རྩིབས་ལའོ། དབང་ལྡན་དུ་ཋཱུ་དང་བདེན་བྲལ་དུ་ཎཱྀ་དང་མེར་ཌྼཱྀ་དང་ས་འོག་ཏུ་ཎཱ་སྟེ། བདེན་བྲལ་གྱི་སྒོ་རུ་ཞེས་པ་གླང་གི་དུས་སྦྱོར་ལའོ། དེ་ནས་འཁྲིག་པ་ལ་སྟེང་དུ་ཎ་སྟེ། ལྷོའི་སྒོའི་དུར་ཁྲོད་དུའོ། ཤར་གྱི་རྩིབས་ལ་ཌྷི་དང་། ལྷོར་ནྪྀ་དང་བྱང་དུ་ཐུ་དང་ནུབ་ཏུ་ནླྀའོ། དེ་ནས་རླུང་གི་འཁོར་ལོ་ལ་ཙའི་སྡེ་པ་ལྷག་པའི་ལྷ་སྟེ། དེའི་ཙླཱྀ་རླུང་གི་རྩིབས་ལའོ། དབང་ལྡན་དུ་ཚཱུ་དང་བདེན་བྲལ་དུ་དྼཱྀ་དང་། མེར་ཛྷཱི་དང་ས་འོག་ཏུ་ཉཱ་སྟེ་མེའི་སྒོ་རུའོ་ཞེས་པ་ཉའི་དུས་སྦྱོར་ལའོ། དེ་ནས་ལུག་གི་དུས་སྦྱོར་ལ་སྟེང་དུ་ཉ་སྟེ་ཤར་གྱི་སྒོའི་དུར་ཁྲོད་ཀྱི་འཁོར་ལོ་ལའོ། ཤར་གྱི་རྩིབས་ལ་ཛྷི་དང་། ལྷོར་ཛྼི་དང་། བྱང་དུ་ཚུ་དང་། ནུབ་ཏུ་ཙླཱིའོ། དེ་ནས་ནམ་མཁའི་ཁམས་ཀྱི་འཁོར་ལོ་ལ་ཀའི་སྡེ་པ་ལྷག་པའི་ལྷ་སྟེ།

ང་ཡི་འཁོར་གྱི་དཾ་པ་བགྱིས། སྟོན་པའི་བཀའ་ལ་ཉན་པར་ཤོག༏ སྲིན་མནན་པའི་འོག་ནས་མ་འཕར་ཅིག། བཅད་པའི་འཕྲོ་ལ་མ་སྐྱེ་ཅིག། ཡང་ནན༏ རཾ་ནན། བྲུཾ་ནན། སྲི་ངན་མ་ར་ཡ་ནན་ནན། འདིའི་བཅའ་ཐབས་ལ། སྲེ་དོང་ནགོ་དཔང་མཐོ་འདོཾ་པ་གང་ལ་བཟོས་པ། ནང་དུ་སྲིན་མོའི་གཟུཊ་དགུའོ། གཅིག་བྱས་ལ་བཞག། མདུན་དུ་ཟས་དང་སྫས་བཀྲཾས། ལྐིང་ཐད་པའཾ་བྲིས་དང་ཐོད་པ་སྤྲད་ལ་སྲི་དོང་དུ་ལྐིང་བརྡུས་ལ། སྲི་འབོད་དུས་མདུན་བསུ་བ་དང་དབལ་བཤོས་ཀྱིས་དབངར་། མནན་དུས་འགར་བ་དང་གཟུཊ་དང་ཁྱི་ཐོད་ནད་དུ་ལྐིང་བསྡུས་ལ་སྲི་དོང་ནང་དུ་མནན། ཁ་ལ་རི་རབས་བཞག་དེ་ནས་མཆྲོེདན་ལྷ་ཁང་ཐེམ་འོགསུ་མནནོ། སྲི་མནན་སྗོགས་ཅེས་བཀྲིས་དགེའོ། སྣ་གོང་དག་ཏུ་ཁྲི་ཅིག་བྲིས། བོན་ཆོས་རུ་྾྾གསུཾ་བྱག་དྱ་ཡིན། འགྲཾ་ངོས་གཡས་ལ། གོད་ཁ་ཐམ་ཅད་ཁ་ཐཾ་བྲིས༏ གཡོན་པ་ལ། གོད་ཁ་ཐཾད་ཆོད་ཅིག། ཁ་ནང་དུ་གོད་སྲི་ཐཾད་ཀུན་མནན་པ་ཡིན་ནོ་མ་ལྡང་ཅིག། བྱས་ལ་ཐེམ་འོག་དྱ་ལཾ་ཡང་ན་སྲང་གི་བཞི་མདོར་མནན། ཟབ་རཾ་ལྠཾ། བཅུས་པ་ལ། ཁ་གཏིང་ཉུལ་པའི་འཇག་མའི་རྩེ་ཟིལ་ཙཾ། འདི་དོན་ལག་ཏུ་བསྟན་ན་རང་གཞན་གྱི་། རྐོམ་ངམ་ནུས་པ་ཡོད་པ་མི་སྲིད་ཀྱང་། བསིལ་ལ་སྐྱོ་བའི་ཚུལ་ཙམ་ཡིན་ནོ་བསྙམ་། ཚེ་འདས་རེ་དོག་ང་བསྲུང་གཞན་དབང་གིས། གྲངས་འབྱཾས་རྣཾས་གཡང་རྐྱང་ལ་ཚེ་བསྐྱལ་ན་། བསྟན་འགྲོའི་ཕན་བདེ་ལྷག་བསཾ་ཡོད་ཅེས་པ། གཉིས་སྐྱེས་ནེ་ཚོའི་སྨྲ་བ་ཁོ་ནར་ཟད། འོན་ཏེ་གང་རེའི་བསྐོར་བའི་དྱལ་ཁཾས་འདིར་། སྔར་ལུང་བོན་གྱི་བསྟན་པ་ཆེས་དར་ཡང་། རྒྱ་གར་བན་དྷེ་འབྱུང་བས་བན་བསྟན་ལྡིང་། ཉི་ཟླ་འཆར་ནུབ་ཚུལ་གྱི་བོན་སྡེ་རྣཾས། རེས་ཤིག་གཏེར་གཤགེས་ཐུགས་གབ་རྣཾ་རོལ་ནི་། འཕགས་པའི་ཚོཊ་ཀྱི་སྟོན་བརྗོད་མི་ལོངས་། མོད་ཁཾས་མེ་ཉག་སྨད་དུ་འགལ་ལུས་རྐྱེན་།

སྨོན་ལམ་རྣམ་པར་དག་པ་བཏབ་པ་ནི། བདག་སོགས་ལུས་ཅན་སྐྱེས་བུ་ནད་མེད་ཅིང་། འཇིག་རྟེན་ཁམས་སུ་ཆར་འབེབས་ལོ་ལེགས་ཤིང་། མི་ནད་ཕྱུགས་ནད་མཐའ་དམག་རྒྱུན་ཆད་ནས། ནད་མུག་མཚོན་གྱི་བསྐལ་པ་ཞི་བར་ཤོག ། ཞེས་བསྔོ་བ་སྨོན་ལམ་གྱིས་རྒྱས་གདབ་བོ། སྨན་ལྷའི་ཆོ་ག་འདི་ནི་ཁྱེའུ་གཉེན་མཐེང་གིས་གཙང་ནུ་མའི་བྲག་གི་གཏེར་མ་མདོ་དྲི་མེད་ཀུན་གསལ་ནང་ནས་ཞང་སྟོན་ལྷུན་གྲུབ་རྒྱལ་པོས་བཏུས་པའོ། ༧ སྨན་ལྷའི་བུམ་སྒྲུབ་ནད་རིགས་འཇོམས་བྱེད་རིལ་བུ་ནི། ༡ ཨ་རུ་ར་ཡི་གཙོར་བྱས་པ། ༢ཇུ་གང་། ༣ གུར་གུམ། ༤ གྷི་ཝཾ། ༥ ཅོང་ཞི། ༦ བྲག་ཞུན། ༧ སྟར་བུ། ༩ དོམ་མཁྲིས། ༡༠ གསེར་མེ། ༡༡སེ་འབྲུ། ༡༢ ཆུ་རྩ། ༡༣ ཐར་ནུ་། ༡༤སྤང་རྩི། ༡༥ ལྕམ་པ། ༡༦ བྲག་སྤོས། ༡༧ པར་པ་ཏ། ༡༨ རེ་སྐོན། ༡༩ གཡའ་ལྕི། ༢༠ ཟངས་རྩི། ༢༡ སྐྱུ་རུ། ༢༢ མ་ནུ་། ༢༤ བུ་རམ། ༢༥ སླེ་ཏྲེས། ༢༦ རྒྱ་མཚོ་ལྦུ་བ། ༢༧ པུ་ཤེལ་རྩེ། ༢༨ ལི་ཁྲི། ༢༩ པི་པི་ལིང་། ༣༠ སུག་སྨེལ། ༣༡ ལི་ཤི། ༣༢ སྤོས་དཀར། ༣༣ཐལ་ཀ་རྡོ་རྗེ། ༣༤ ཙན་དན། ༣༧ ཛཱ་ཏི། ༣༨ ཨ་ཀ་རུ། ༣༩ ཤིང་ཚ། ༤༠ སྡིག་སྲིན། ༤༡ སྒོག་སྐྱ། ༤༢ ཏིག་ཏ། ༤༣ ཨ་བྱག ། ༤༤ གུ་གུལ། ༤༥ ཤུ་དག ། ༤༦ སྒ་སྐྱ། ༤༧ སྒ་སྨུག ། ༤༨ དུག་མོ་ཉུང་། ༤༩ཤིང་ཀུན། ༥༠ ཞོ་ཤ་གསུམ། ༥༡ རྒྱ་མཚལ། ༥༢ སྲོ་མ་ར་ཙ། ༥༣ བ་ཤ་ཀ། ༥༤ ཞུ་མཁན། ༥༥ པྲི་ཡང་ཀུ། ༥༦ བྱ་རྒོད་སྤོས། ༥༧ རྒྱ་སྐག ། ༥༨ ལྕགས་ཀྱུ་བ། ༥༩ མིང་ཅན། ༦༠ སྟག་ཤ། ༦༡ ཤེལ་དཀར། ༦༢ སྤང་རྒྱན། ༦༣ བྲག་ཇ། ༦༤ བོང་ང་། ༦༥ རྒྱ་ཚྭ། ༦༦ ཁ་རུ་ཚྭ། ༦༧ གཟེ་མ། ༦༨ སྦྲུལ་ཤ། ༦༩ གླ་རྩི། ༧༠ ག་པུར། ༧༡རིན་ཆེན་སྣ་ལྔ། དེ་རྣམས་སྦྲང་རྩི་སྦྱར་བྱས་ལ། རིལ་བུ་ཤྭ་བའི་རིལ་མ་ཙམ་དུ་བྱས། རིན་ཆེན་ལྷུང་བཟེད་ནང་དུ་བླུགས། སྒྲུབ་ཁང་དབུས་སུ་བཀོད་པར་བྱ།

རྡོ་རྗེ་བདུན་གྲུབ་ཐེག་དགུའི་ཡོན་ཏན་རྫོགས། གང་ལ་གང་འདུལ་སྣ་ཚོགས་ནོར་བུ་བཞིན། བདག་གཞན་རེ་སྐོངས་མཆོག་གི་དངོས་གྲུབ་སྩོལ། སྐུ་གསུམ་དབྱིངས་གཙོ་ཀུན་ཏུ་བཟང་མོ་ནི། མཚོ་རྒྱལ་སྒྲོལ་མ་གསང་ཡེ་ཁྲོས་མ་ནག ། ཕྱི་ནང་གསང་བ་ཡང་གསང་དཀྱིལ་འཁོར་གཙོ། རིགས་ལྔ་རིགས་བརྒྱ་འབུམ་ཕྲག་མཁའ་འགྲོ་མ། ཐམས་ཅད་འདུས་པའི་དཀྱིལ་འཁོར་ལྷ་ཡི་གཙོ། བཀའ་གཏེར་སྲུང་མ་བཅས་ལ་གསོལ་བ་འདེབས། ཕྱི་ནང་གསང་བའི་བར་ཆད་དབྱིངས་སུ་གསོལ། ཉམས་ཆག་འགལ་འཁྲུལ་ནོངས་པའི་བཀའ་ཆད་ཟློག ། རྩ་ཐིག་རླུང་དང་ཕྲིན་ལས་ནུས་སྟོབས་སྤོར། འདུ་དང་འབར་དང་མཆོག་གི་དངོས་གྲུབ་སོགས། མཆོག་ཐུན་རབ་འབྱམས་ལས་ཀྱི་མཐུ་དབང་སྩོལ། བསྐྱེད་རིམ་ལམ་གྱི་སྣང་ཡུལ་ལྷ་རུ་འཆར། རྫོགས་རིམ་ནུས་པས་ངག་རླུང་ནཱ་དར་ཐིམ། ཐབས་གྲོལ་ཟབ་མོས་རིག་སྟོང་ཆོས་སྐུར་སྨིན། བཞི་པའི་ལམ་མཆོག་ཆོས་ཉིད་ཟད་སར་འཁྱོལ། ནང་གསལ་བུམ་པ་སྐུ་རུ་འུབ་ཆུབ་སྟེ། ཕྱིར་ཤར་བྱ་རྩོལ་བྲལ་བའི་ཕྲིན་ལས་ཀྱིས། འགྲོ་འདུལ་འཁོར་བ་དོང་སྤྲུགས་མཐུ་ཐོབ་ཤོག ། ཅེས་པའང་དགེ་སློང་འཇམ་དཔལ་ཚུལ་ཁྲིམས་ནས་ཆེད་དུ་བསྐུལ་བ་བཞིན། ༧འཇམ་མགོན་ཆོས་ཀྱི་རྒྱལ་པོའི་བཀའ་དྲིན་ཁོ་ནས་འཚོ་བའི་འབངས། ཀུན་བཟང་མཁའ་ཁྱབ་རང་བྱུང་བདེ་བའི་རྡོ་རྗེར་འབོད་པའི་སྔགས་བཙུན་གྱི་རྟགས་ཙམ་འཛིན་པ་དེས་གུས་པ་ཆེན་པོས་སྦྱར་བའོ། ཚིག་བདུན་སྒྲུབ་ཐབས་ཟབ་མོའི་སྨིན་གྲོལ་ལག་ལེན་མཚམས་སྦྱོར་གྱིས་བརྒྱན་པ་པདྨའི་ལྡིང་ཁང་ཞེས་བྱ་བ་བཞུགས་སོ། ཨོཾ་སྭསྟི། ཐར་འདོད་སྐྱེས་བུ་རྣམ་གྲོལ་དགའ་ཚལ་དུ། འདྲེན་པའི་དེད་དཔོན་ཨོ་རྒྱན་བླ་མ་དང་། དབྱེར་མེད་རྗེ་བཙུན་པདྨ་གར་གྱི་དབང་། མི་འབྲལ་བདེ་ཆེན་འཁོར་ལོར་རྟག་ཏུ་རོལ། མཁའ་འགྲོའི་ཐུགས་ཀྱི་གསང་མཛོད་རྡོ་རྗེའི་གནས།

སྔོན་བསགས་ལེགས་སྨོན་ཚུལ་ཁྲིམས་རྣམ་པར་དག་པའི་ཞིང་ས་གཤིན་པོ་ལ། རབ་དཀར་དགེ་བཅུའི་ས་བོན་བཏབ་གང་དེང་འདིར་དུས་སུ་སྨིན་པ་ལས། རྨད་བྱུང་དལ་བ་བརྒྱད་ཀྱི་སྙེ་མ་འབྱོར་བཅུའི་འབྲས་བཟང་གཡུར་སྨིན་པ། འདི་འདྲ་བསྐལ་པ་བརྒྱར་ཡང་རྙེད་དཀའ་གྲོགས་དག་བག་མེད་མ་བྱེད་ཅིག ། མི་ཚེའི་ཉིན་བྱེད་སྐད་ཅིག་མི་སྡོད་ཆུ་ལྷའི་ས་འཛིན་རྩེ་མོར་གཞོལ༑ ༑ལང་ཚོའི་མེ་ཏོག་ཡུན་དུ་མི་བརྟན་རྒ་དང་ན་བའི་བ་མོས་བཅོམ། སྐྱེ་བའི་ལྗོན་ཤིང་འདི་ནི་འཆི་བདག་སྟ་གྲི་འཛིན་པའི་ཁར་གནས་པས། ཀྱེ་ཧུད་སོས་དལ་བྱེད་པ་འདི་རྣམས་ངེས་པར་ཕྱུགས་ལས་བླུན་པར་སེམས། ས་དང་ལྷུན་པོ་རིར་བཅས་འབར་ཞིང་གཡོ་བས་མཁའ་ལ་འགྲོ་སྲིད་ཀྱང་། ཀུན་མཁྱེན་ལེགས་གསུངས་རྒྱུ་འབྲས་ཟབ་མོའི་ཚུལ་ལ་བསླུ་བ་ནམ་ཡང་མིན། དེ་སླད་དགེ་ལེགས་རྩི་ཤིང་སྲེག་བྱེད་སྡིག་པའི་མེ་ཚོགས་རབ་གཞིལ་ལ། དམ་པ་གསུམ་ལྡན་དགེ་ཚོགས་ཆར་གྱིས་རྣམ་གྲོལ་མྱུ་གུ་བསྐྱེད་པར་མཛོད། མ་རིག་གཟུང་འཛིན་འཛིན་ཁྲིས་དབང་བྱས་སྡུག་བསྔལ་གསུམ་གྱི་རྦ་རླབས་ཅན། ཐར་དཀའ་གཏིང་ཟབ་སྲིད་མཚོ་འདིར་བྱིངས་ཐོག་མེད་དུས་ནས་ཉམ་ཐག་པས། ད་རེས་དལ་འབྱོར་གྲུ་རྙེད་མཚན་ལྡན་དེད་དཔོན་མཁས་པས་རྗེས་བཟུང་ནས། གདམས་ཟབ་སྐྱ་བ་ལེགས་པར་ཐོབ་དུས་བརྩོན་པས་ཕ་རོལ་བརྒལ་བར་མཛོད། ཐུབ་བསྟན་མ་དྲོས་འགྲོ་བསོད་རྫིང་བུར་འཁྱིལ་བའི་སྒྲུབ་བརྒྱུད་ཤིང་རྟ་བརྒྱད། ཀུན་ཀྱང་ལུང་རྟོགས་འཛུམ་དཀར་གྲོལ་ལ་མཁས་གྲུབ་ངང་པའི་རྒྱལ་པོ་རྣམས། བཤད་སྒྲུབ་ཕྲིན་ལས་ངོ་མཚར་བྱེ་བའི་ཟློས་གར་བསྒྱུར་བའི་དཔལ་གྱིས་རྩེན། མཐར་ཐུགས་ངེས་དོན་རྒྱ་མཚོར་རོ་གཅིག་དེ་སླད་དག་པའི་བློ་མིག་སྦྱོངས། དྲི་མེད་ཐོས་བསམ་ཡོན་ཏན་ཤིན་ཏུ་རྒྱ་ཆེ་ལུང་གི་མཁའ་དབྱིངས་སུ། ཟབ་གསང་ཉམས་མྱོང་རྟོགས་པའི་ཆུ་འཛིན་ཕྲེང་བས་ཕྱོགས་ཀྱི་མཐར་ཁྱབ་ལས།

མེ་མེ་ལ་ཐིམ་པས་ལུས་ཀྱི་དྲོད་སྡུད། རླུང་རླུང་ལ་ཐིམ་པས་དབུགས་ངར་ཡན་ལག་ཕྲ་ཞིང་མིག་གྱེན་དུ་བཟློག་བཟའ་བཏུང་ཁོང་དུ་མི་ནུས་པ་དབུགྃས་ཕྱིར་འཕེན་པའི་ངརྃ་སྒྲྃ་དང་བཅས་བརྩེྃག་མར་འབྱིན། འྃདེགས་བྱེྃད་ཀྱི་རླུངྃ་བྲོྃས་ཏེ་མི་གནས་པས་ལུས་ཡན་ལག་གིས་མི་ཐེག ། རྐང་ལག་བརྐྱང་བསྐུམ་མི་ནུས། ལུས་ལ་གཤིན་དྲི་མནམ་ཞིང་ལུས་ཀྱི་ཆ་མེད་པ་དཀར་ནག་གི་ལམ་ནས་ཁ་སྣར་ཐོན་ཏེ་དབུགས་ངརྃ་པོའི་སྐདྃ་ཅནྃ་གྱིྃ་སྒྲ་དྲག་པོ་འབྱིན་པའི་ང་རོྃས། གཉེན་ཉེྃ་འབྲེལྃ་བུ་ཚ་ཕ་མ་ཀུན་གྱིས་ཤི་འབྲལ་དུ་འབྱུང་བ་ཤེས་ནས་གདུངྃ་སེམྃས་ཀྱིས་འབྲལ་མི་ཕོད་ཀྱང་བྱ་བའི་ཐབས་ཟད། བརྩེ་བའི་འབྲེལྃ་ཐྃག་གཤིན་རྗེའི་ཞགས་པས་དྲངས་ཏེ་གཅོྃད་པ་ཡང་། ཆགས་སེམས་ཀྱི་དབང་གིས་མིག་ནས་མཆི་མ་དང་སྨྲེ་སྔགས་ཀྱི་སྒྲ་འདོན་བཞིན་འཇིག་རྟེན་ཕྱི་མ་ཆ་མེད་ཀྱི་ཡུལ་དུ་རྒྱུས་མེད་གཅིག་པུ་འགྲོ་བྃའི་ཚེྃ་འཆི་རྐྱེན་མཚོན་གྲི་སོགས་མ་རུངས་པ་མི་འབྱུང་ཞིང་སྲོག་གི་གནྃད་གཅོྃད་པའམ་ཆད་པས་རླུང་སེམས་འབྲེལ་ཆད་དེ་སེམས་དང་ཡེ་ཤེས་རྒྱབ་གྱེས་པའི་ཚེ་ཚོར་བའི་སྡུགྃ་བྃསྔལ་དྲྃག་པོྃ་བཟོད་པར་དཀའ་བས་ལུས་སེམས་ཉམ་ཐག་པ་མིྃ་འབྱུངྃ་ཞིངྃ་རྣམ་ཤེས་རླུང་ཞོན་གྱི་གདམས་ངག་ལ་བརྟེན་ནས་བར་དོ་མེད་པ་འཆི་ཁར་དཔའ་བོ་མཁྃའ་འྃགྲོའི་ཚོགས་ཀྱིས་གདུགས་དང་རྒྱལ་མཚན་བསྒྲེངས་ཤིང་རོལ་མོའི་སྒྲ་དང་བཅས་ཏེ་རིགས་ཀྱི་བུ་གང་འདོད་པའི་དག་པའི་ཞིང་ཆེན་པོར་བཞུད་པའི་དུས་ལ་བབ་ཞེས་བསུྃ་བའིྃ་སྣངྃ་བྃ་དག་པར་འཆྃར་བས་བུ་ཕའི་ཁྱིམ་དུ་འགྲོ་བ་ལྟར་དགའ་སྤྲོའི་ཉམས་དང་བཅས་ཏེ་སྐྱེ་འཆིའི་སྡུག་བསྔལ་གཏན་ནས་མེད་པར་དག་པའི་ཞིང་ལ་དབུགས་དབྱུང་བརྃ་ཤོྃག་ཅིག ། གཉིས་པ་འཆི་ཁ་བར་དོ་ཆོས་སྐུའི་གཞི་ཐོག་ཏུ་གྲོལ་བར་སྨོན་པ་ནི།

Rare stacks: ཆུྃ ཆེྃ ཉིྃ ཟླྃ སྒྱུྃ
བཟློག་མེད་དུས་ཀྱི་ཞགས་པས་ཐེབས་པ་ལ་ཡ་ང་བ་དང་འཇིགས་སྐྲག་དུ་མས་གཙེས་པས་ཉམ་ང་བ་ལ་ཀྱིྃ་ཧུདྃ་ཅེས་སྐྱོ་བའི་ཚིག་གིས་བསྐུལ་ནས། སྡུག་བདེན་གྱི་མཚན་གཞི་ལུས་འདི་གྲུབ་པ་ཙམ་ནས་སྃ་ལས་ཤ་རུས་སོགས་སྲ་ཞིང་ལྗིད་པ། ཆུྃ་ལས་ཁྲག་དང་ཆུ་སེར་སོགས་གཤེར་བ། མེྃ་ལས་དྲོད། རླུངྃ་ལས་དབུགས། བུ་ག་ལས་ནམྃ་མཁྃའི་ཁམས་ཀྱི་འབྱུངྃ་ལྔའིྃ་ནུས་པ་མཐུན་ཕྱོགས་སུ་ཚོགས་པས་བསྐྱེད་དེ་ནུས་སྟོབས་ཡར་འཕེལ་གྱི་ཚད་རྫོགས་ནས། ཁྱབ་བྱ་འབྱུང་བའི་ཁམས་དང་ཁྱབ་བྱེད་ཡེ་ཤེས་ཀྱི་ཐིག་ལེ་རྒྱབ་གྱེས་ཏེ་སོ་སོར་ཁ་ཕྲལ་ནས་ཡུལ་ལྔ་ཡེ་ཤེས་ལ་ཐིམ་པར་ཕྱོགས་ཚེ་ཕྲ་བའི་དབྱིངས་སུ་ཐིམྃ་པའི་རྟགས་སྣང་རིམྃ་གྱིས་འཆར་ཏེ་རགས་པ་ལུས་གནས་ཀྱི་རྩ་དང་རླུང་ཆད། ཐིག་ལེ་གནས་ནས་འཕོ་བས་ཚོགས་པ་གྱེས་ཏེ་ལུས་སེམས་འབྲེལ་ཆད་གདོས་བཅས་ཀྱི་ཕུང་པོ་ཤུལ་དུ་བོར། རྣམ་ཤེས་ཁོ་ན་ལས་ཀྱི་རྣམ་སྨིན་དང་འགྲོགས་ནས་དགེ་སྡིག་གི་འཕེན་པ་གང་ཡོད་དུ་འགྲོ་བར་ཆས་པའི་ཚེ་རླུང་ཐིག་སེམས་གསུམ་དབུ་མར་ཐིམ་པའི་རྟགས་སྣང་དུྃ་བྃ་དང་སྨིགྃ་སྒྱུྃ་གཡོ་བ་ལྟ་བུ་དངྃ། སྲིན་བུ་མེྃ་ཁྱེརྃ་ལྟ་བུ་དང་། མརྃ་མེའིྃ་འོད་ཁྱིམ་ལྟ་བུའི་སྣངྃ་བྃ་རྣམས་གསལྃ་བའིྃ་རགས་པའི་རྩ་རླུང་ཐིག་ལེ་དྭངས་མའི་གནས་སུ་ལྡོག་པས་དུས་སྦྱོར་བཅུ་གཉིས་ཀྱི་རླུང་འགག་སྟོབས་ལས་དེ་རྗེསྃ་སུ་ཕྲྃ་བའིྃ་དཀར་དམར་སེམས་གསུམ་གནས་འཕོ་སྟེ་ཐིམྃ་རིམྃ་གྱི་ཕྲ་བ་སྣངྃ་མཆེྃད་ཐོབྃ་པའི་སྣང་བ་སྐུ་གསུམྃ་གྱི་རང་སྣང་དུ་ངོ་ཤེས་པས་རིག་པ་རང་ཐོག་ཏུ་མཉམ་པར་བཞག་པའི་དབྱིངས་སུ་གྲོལ་བར་སྨོན་པ་སྟེྃ། འདིྃ་ལྟརྃ་ལུས་ཀྱི་དྭངས་མ་ཐིག་ལེ་དམར་ཆ་ཨ་ཐུང་གི་རང་བཞིན་ལྟེ་བ་ནས་སྙིང་གར་འཕོ་བའི་ཚེ་རྣམྃ་ཤེསྃ་སྣངྃ་བྃ་ལྃ་ཐིམྃ་པསྃ་ཕྱི་རྟགས་སུ་སྤྲིནྃ་མེདྃ་དྭངས་པའི་ནམྃ་མཁའ་ལྃ་ཉིྃ་ཟླྃ་གཟའྃ་སྒྲ་གཅན་གྱི་ཁ་ཡིས་ཟིནྃ་པ་ལྟརྃ་དམརྃ་ལམྃ་གྱི་སྣང་བ་འཆརྃ་ཞིངྃ་མ་ལས་ཐོབ་པའི་དམརྃ་ཆྃ་ཁྲག་གི་དྭངས་མ་སྙིངྃ་གརྃ་ལྡོགྃ་སྟེ་ཐིམ།

རྫི༷་ཕྱོག༷ས་མ༷་ཡིན༷་པར་ལྔ༷་བཅུ༷ར་ཁྱབ༷་པའི༷་ལྷ་རྣམས་ཀྱི་འདོད་པའི་ཡོན་ཏན་ཡིད་ལ་བསམ་པ་དང་མཉམ་དུ་འབྱུང་བའི་ཤིང་ཡིན་ལ། དེའི་དྲུང༷་དུ་རིན་ཆེན་ཨ༷ར་མོ༷་ནིག༷་དང་འདྲ༷་བ་རབ་ཏུ་དཀར་བའི༷་རྡོ༷་ལེབ༷་མཛེ༷ས་པ༷་སྟེ༷། དེའི་ཚད་ཆུ༷་དང་ཞེང༷་དུ་དཔ༷ག་ཚད༷་ལྔ༷་བཅུ་ལྔ་བཅུའི༷་རང༵་བཞི༷ན་གྲུ་བཞི་པ་མཐའ་སྐོར་དུ་ཉིས་བརྒྱ་དང་འཕང་དུ་༼འཕངས་སུ༽དཔག་ཚད་ཕྱཻད་དང་དྲུག་དང་བཅོ་བརྒྱད་ཀྱི་ཆ་གཅིག་གིས་ལྷག་པ་ལ༵། དབུས་ན་དབ༷ང་པོ༷འི་ཁྲི་དང་ལྷ་གཞན་རྣམས་ཀྱི་ཁྲིར༷་བཅ༷ས་བཀོད་པས་མཛེས་ཤིང་མེ༷་ཏོག༷་གི་ཆ༷ར་པ་འབ༷བ་པར༷་ལྷ་རྣམ༷ས་དབྱར༷་དུས༷་སུ༷་ཟླ༵་བ༷་བཞི༷ར་འདོད་ཡོན་ལ་ལོངས་སྤྱོད་ཅིང་གནས༷་སོ༷། ལྔ་པ་འདུན་ས་ནི༑གྲོང་ཁྱེར་བལྟ༷་ན༷༷་སྡུག༷༷་ན༷ས༷་དཔ༷ག་ཚད༷་ཕྱེད༷་དང༷་བཞི༷འི་གཞལ་བའི་ལྷོ༷་ནུབ༷༷་ན་ཆོས༷་བཟང༷་ལྷ༷་ཡི༷་འདུན་ས་སྟེ། ཆུ་ད༷ང་ཞེ༷ང་དུ་དཔག་ཚད་སུམ༷་བརྒྱ༷་མཐ༷འ་སྐོར༷་དུ་དགུ༷་བརྒྱ༷་དབྱིབས་ཟླུམ༷་པོ། དབུས༷་ན༷་རི༷ན་ཆེ༷ན་གྱི་ཁང༷་བརྩེག༷ས་འོད༷་འབ༷ར་བ༷། ཅུ༷ང་ཟ༷ད་ཙམ༷་གྱིས༷་ས༷་ལ༷་མ༷་རེག༷་པར་བཏེགས་པའི་ཁང་བུ་དོན་ཐམས་ཅད་གྲུབ་པར་བྱེད་པ་ཞེས་བྱ་བའི་དྲུང༷་དུ་ལྷ༷་རྣམ༷ས་འདུས་ནས་རང༷་གི༷་སྲིད༷་དང༷་རང་རང་གི་བྱ༷་བ༷་ལས༷་ཉམ༷ས་པ༷་བསྲུ༷ང་བའི་ཕྱིར༷་འདུས་སར་འགྲོ༷་བའི༷་གནས༷་ཡོད༷་དོ༷། དེས་ས་གཞི་ལ་བརྟེན་པའི་གནས་བཤད་ཟིན་ནས། གཉིས་པ་ནམ་མཁའ་ལ་བརྟེན་པ་ནི། སུམ་ཅུ་རྩ་གསུམ་གྱི་སྟེང་གི་ནམ༷་མཁ༷འ་ལ༷་བརྟེ༷ན་པའི་འདོ༷ད་ལྷའི༷་གནས༷་འཐབ་བྲལ་དང་དགའ་ལྡན་དང་འཕྲུལ་དགའ་དང་གཞན་འཕྲུལ་དབང་བྱེད་བཞི༷་པོ་ནི༷། རྒྱུ་ཟག་བཅས་བསོད་ནམས་ཆ་མཐུན་གྱི་དགེ་བས། དུས་ནི་སུམ་ཅུ་རྩ་གསུམ་པའི་སྔར་ཆགས་ལ། གནས་ཁང་ནི་གཟ༷འ་སྐར༷་བཞི༷ན་དུ༷་ནམ་མཁ༷འ་ལ༷་སྣང༷་བ༷་སྟེ༷།

ཟ་འཐུང་ཕུད་ཀྱིས་མཆོད་དང་གསོལ་བཏབ་ན། འ་འུར་བྱིན་རླབས་འཕོ་བའི་དབང་བསྐུར་གྱིས། ཡ་མཚན་རྟོགས་པའི་རང་ཞལ་མཇལ་བར་ངེས། ར་བ་བརྒལ་དཀའ་སྲིད་པའི་གྲོང་གནས་འགྲོ། ལ་འུར་བདེ་ལྡན་ཞིང་དུ་དབུགས་དབྱུང་ཕྱིར། ཤ་ཟན་གདོལ་ཕྲུག་བརྩོན་པའི་གོ་བགོས་ལ། ས་བཅུ་ལམ་ལྔའི་མངོན་རྟོགས་ངོམ་པར་མཛོད། ཧ་ཧ་གཅེན་གྱི་ཉམས་མཚར་བསླབ་བྱའི་ཕྲེང་༑ ༑ཨ་གྲོ་བན་ཆུང་དགྱེས་པའི་གསོས་སུ་འབུལ། ཁྱེད་བློ་ཐོས་བསམ་སྒོམ་པས་རབ་ཕྱུག་སྟེ། འཆད་རྩོད་རྩོམ་པའི་ལང་ཚོ་དར་གྱུར་ཅིག ། ཅེས་པའང་རང་གི་གཅུང་དགེ་ལེགས་རྒྱ་མཚོ་ལ་ཕན་སེམས་གོ་འདོད་ཀྱི་ཚུལ་དུ་གདམས་པའོ། འདི་དང་གཏན་དུ་དགེ་བའི་བསླབ་བྱ། སྐྱབས་ཀུན་འདུས་ཞལ་དྲིན་ཆེན་ཆོས་ཀྱི་རྗེ། འབྲལ་མེད་སྙིང་དབུས་པདྨོར་རྟག་བཀོད་ནས། འདི༵་དང༵་གཏ༵ན་དུ༵་ད༵གེ་བ༵འི་བསླབ༵་བྱ༵འི་ཕྲེང་། ཁྱེད་གསུང་མི་ལྡོག་ཙམ་དུ་མདོ་ཙམ་བྲི། དལ་འབྱོར་རྙེད་དཀའ་འབུངས་གསུམ་ཆོས་ལ་ཐོན། ནམ་འཆི་ཆ་མེད་བརྩོན་འགྲུས་ལྕག་གིས་བྲོབས། རྒྱུ་འབྲས་བསླུ་མེད་བླང་དོར་ཚུལ་བཞིན་གྱིས། འཁོར་བའི་སྡུག་བསྔལ་བཟོད་དཀའ་སྐྱོ་བ་སྐྱེད། ནམ་ཡང་མི་བསླུ་མཆོག་གསུམ་སྐྱབས་སུ་བསྟེན༑ ༑ཡོན་ཏན་གཞི་རྟེན་ཚུལ་ཁྲིམས་རྣམ་དག་སྲུངས། ཐེག་ཆེན་རྩ་བ་བྱང་ཆུབ་སེམས་མཆོག་བསྐྱེད། བློ་སྦྱོང་སྙིང་པོ་བྱམས་དང་སྙིང་རྗེ་སྒོམས། གསང་སྔགས་ལམ་སྲོག་དམ་ཚིག་རྣམ་དག་བྱོས། བྱིན་རླབས་རྩ་བ་བླ་མར་མོས་གུས་རེམས། དངོས་གྲུབ་གཞི་མ་ཡི་དམ་བསྙེན་སྒྲུབ་འབུངས། ཕྲིན་ལས་ཐོག་མེད་མཁའ་འགྲོ་ཆོས་སྐྱོང་མཆོད། ཁྱད་པར་དབང་རྣོན་ཚེ་འདིར་གྲོལ་བའི་ཐབས། བླ་མ་སངས་རྒྱས་དངོས་སུ་མཐོང་བའི་བློས། གསོལ་བཏབ་དབང་བླངས་ཐུགས་ཡིད་དབྱེར་མེད་བསྲེས། བློ་བྲལ་རྫོགས་པ་ཆེན་པོའི་དགོངས་པ་སྐྱོངས།

བསང་ཚིག་ཡི་གེར་བྲི་ཞིང་སྟོན་འདི་ལ། བར་གཅོད་བདུད་དགྲའི་བསམ་སྦྱོར་རྩད་ནས་ཐུལ། ལས་བཞིའི་སྒོ་འབྱེད་འགྲོ་ཕན་མཐའ་རྒྱས་མཛོད། ཐབས་མཆོག་བདེ་བའི་ཡང་སྙིང་འཁོར་ལོ་སྡོམ། འཛམ་གླིང་སྐྱེ་དགུར་སྙན་བརྒྱུད་བདུད་རྩིའི་སྲོལ། འབྱེད་སླད་བྲམ་ཟེའི་རིགས་འཁྲུངས་གྲུབ་པའི་ཏོག ། ཏཻ་ལོ་པྲཛྙཱ་བྷ་དྲར་གུས་ཕྱག་འཚལ། བླ་མའི་བཀའ་བཞིན་ལུས་སྲོག་ལྟོས་མེད་དུ༑ ༑གཏོང་བའི་དཀའ་སྤྱད་མཐར་ཕྱིན་དབང་བཞི་ནོད། གདམས་པ་ཟབ་གསལ་ཟུང་འཇུག་གསལ་བྱེད་སྒྲོན། ནཱ་རོ་པཎ་ཆེན་འཛམ་གླིང་མུན་སེལ་མཆོག ། དེ་ཡི་ཞལ་ལུང་སྙན་བརྒྱུད་སིདྡྷི་ཞེས། གང་ཐུགས་རཏྣའི་སྣོད་དུ་ལེགས་གཏམས་པས། བསིལ་ལྡན་འགྲོ་བའི་ལས་ཉོན་ནད་སེལ་སྨན། སྦྱིན་མཁན་མར་སྟོན་བཀའ་དྲིན་ཕ་མཐའ་ཡས། འཁོར་བའི་བདེ་འབྱོར་ཀུན་ལ་རྨི་ལམ་བཞིན། གཟིགས་ཏེ་ཡལ་དོར་དབེན་གསུམ་ཏིང་འཛིན་བརྟན། རླུང་སེམས་དྷཱུ་ཏིར་རྣམ་དག་གྲུབ་པའི་དབང་། མི་ལ་རྗེ་ལྟར་བརྒྱུད་འཛིན་ཀུན་འགྲུབ་ཤོག ། ཐུབ་པས་ལུང་བསྟན་དགེ་སློང་འཚོ་བྱེད་ཞབས༑ ༑འོད་གསལ་ཕྱག་ཆེན་གདམས་པས་སྐལ་ལྡན་རྣམས། མི་སློབ་ལམ་མཁན་ཐུགས་རྗེ་སྣང་བའི་མཛོད། སྒམ་པོ་པ་དེས་ངེས་དོན་པད་ཚལ་སྐྱོངས། ཉམས་རྟོགས་ཆ་ཤས་ཡོང་གང་མཁྱེན་བརྩོན་གྱི། རི་དྭགས་རྟགས་ཅན་སྨིན་གྲོལ་འོད་དཀར་གྱིས། འགྲོ་རྒྱུད་མཚན་མུན་གཞིལ་མཁན་རས་ཆུང་དང་། ངམ་སྟོན་བརྒྱུད་པར་བཅས་པས་བྱིན་ཆེན་ཕོབ། སྦྱིན་སོགས་ཕྱིན་དྲུག་ལམ་གྱི་༄༅། ཚིགས་པ་ཅན། ཟབ་གསལ་ལྷ་སྔགས་སྒྱུ་འཕྲུལ་ཤུན་པས་མཛེས། མཆོག་དངོས་འབྲས་བུར་ལྡན་པ་མརྐ་ཊེ། འགྲོ་མགོན་ཕག་གྲུའི་མཚན་སྙན་ནོར་འཛིན་ཁྱབ། མི་འཁྲུགས་འཇིག་རྟེན་དབང་སོགས་རྒྱལ་ཀུན་ཆོས་སྐུའི་ལྷ་ལམ་ཡངས་པོར་རྒྱུ། ཚུལ་ཁྲིམས་ཏིང་འཛིན་ཤེས་རབ་ལ་སོགས་སྲིད་ཞིའི་ཡོན་ཏན་ཆ་ཀུན་རྫོགས།

སྣ་ཚོགས་ཐེག་པ་ཀུན་དང་རིམ་པ་ལྔ། བདེ་གསང་འཇིགས་དུས་ལ་སོགས་ཕ་མའི་རྒྱུད། དུ་མའི་རིམ་གཉིས་གནད་ཀུན་འདིར་འདུ་ཞིང་། མདོ་འདུལ་མངོན་སོགས་ཐུན་མོང་དགོས་པའང་འགྲུབ། དེ་ཕྱིར་གཞུང་མང་ཐོས་བསམ་དཀའ་ཚེགས་ལ། མ་ལྟོས་གཅིག་ཆོག་རྒྱལ་པོ་བསྲེ་འཕོའི་གནད། གཅིག་ཤེས་ཀུན་གྲོལ་འདི་ཉིད་རབ་གསང་སྟེ། སྐལ་ལྡན་བྱང་ཆུབ་འདོད་རྣམས་ཉམས་སུ་ལོངས། རྒྱལ་ཀུན་རིགས་བདག་རྡོ་རྗེ་འཆང་དབང་གསུང་། ཡེ་ཤེས་མཁའ་འགྲོའི་དཀོར་མཛོད་གཉན་པོ་ནས། གྲུབ་མཆོག་ཏཻ་ལོ་ཕྱག་རྒྱ་སྟངས་སྟབས་ཀྱིས། དབང་མེད་ཕྱུང་སྟེ་འཛམ་གླིང་འཕགས་ཡུལ་དང་། དེ་ཡི་བཅུད་མཆོག་ཟབ་དོན་སིདྡྷི་ཞོ། མངའ་བདག་ནཱ་རོའི་ཐུགས་མཆོག་བཻཌཱུརྱར་སྣོད། ལེགས་གཏམས་བྷོ་ཊའི་མ་རིག་ལིང་ཏོག་ཀུན། སེལ་མཁས་མར་སྟོན་ཚེ་རིག་མཁན་དེས་ཐོབ། དེས་ཀྱང་མཁའ་འགྲོའི་ལུང་བསྟན་མི་ལ་རྗེར། ཡོངས་གཏད་འགྲོ་མང་སྲོག་འཚོའི་གནང་བ་ཐོབ། དྭགས་རས་ངམ་སོགས་གདམས་པའི་སྨན་རྐྱལ་ཁུར། ཉམས་མྱོང་ཐུར་བཙག་ཕྱག་ན་བསྲེལ་བ་འགས། ལས་ཉོན་ཟུག་རྔུ་སེལ་བྱེད་ལོག་གནོན་མཆོག ། སྙན་བརྒྱུད་རིན་ཆེན་སྦྱར་བས་ལེགས་བསྲེས་ཏེ། སྒོམ་བྱུང་མྱོང་བའི་འདམ་བཏགས་ཞིབ་མོར་བཏུལ། ཁྲིད་ཡིག་སྙིང་ཁུ་འདུས་པའི་རིལ་བུར་དྲིལ། དེ་སླད་སྔོན་བྱོན་མཁས་གྲུབ་དུ་མ་ཡི། མ་ནོར་མན་ངག་པུསྟ་ཀའི་ཕྲེང་བ། མང་དུ་སྤེལ་བ་གཏིང་ཟབ་ཆུ་གཏེར་ལས། ཉིང་ཁུ་བདུད་རྩིའི་ཐིགས་པ་འདི་ན་མཆིས། བདག་འདྲ་མང་ཐོས་ནོར་དང་བྲལ། བསམ་སྒོམ་བག་ཆགས་མཐུ་ཕྲ་བས། ཉམས་མྱོང་དྲོད་རྟགས་ཆ་ཙམ་ཡང་། མ་ཐོབ་མཁས་པའི་བཤད་སྦྱར་དང་། གྲུབ་པའི་མན་ངག་གནད་འདོམ་པའི། ནུས་པ་རི་བོང་རྭ་དང་མཚུངས། དེས་ན་སྔོན་མེད་ལེགས་བཤད་ཀྱི། དགའ་སྟོན་ཅི་ཞིག་༄༅། འཇོ་བར་ནུས། འོན་ཀྱང་སྔོན་བྱོན་མཆོག་གི་གསུང་།

ཀུན་གྱི་མིག་གཉིས་བྷནྡྷ་རུ། རང་བྱུང་མཆི་མའི་སྨན་རཀ་བསྐྱིལ། བཞིན་གྱི་ལྷ་སྐུ་མཆོད་ན་ཡང་། དོན་གོ་སྣ་གཅིག་ཨེ་ཡོད་སོམས། དེའི་ཚེ་ལུས་ཀྱི་ཁང་རུལ་ནས། རྣམ་ཤེས་སྐྱེས་བུ་ཕྱི་རུ་ཐོན། ལས་ཀྱི་རླུང་གི་རྟར་ཞོན་ནས། གྲོགས་མེད་གཅིག་པུར་ཕྱི་མར་འགྲོ། དེ་དུས་རང་ལ་ཕན་ངེས་པ། དཀར་པོ་དགེ་བ་མ་གཏོགས་མེད། ཡུན་རིང་འགྲོགས་པའི་གཉེན་དང་འབྲལ། བསགས་པའི་ཟས་ནོར་ལོངས་སྤྱོད་ཚོགས། ཏིལ་འབྲུ་ཙམ་ཡང་རྗེས་མི་འབྲང་། དེ་ལྟར་འགྱུར་བར་ཤེས་བཞིན་དུ། ཕྱི་མའི་བྱ་བ་ཇི་ལྟར་བཤོལ། ཚེ་འདིར་ཡུན་རིང་སྡོད་ལོང་མེད། ལེ་ལོ་ཇི་ལྟར་བྱ་བར་རིགས། ཤ་རུས་དག་ཀྱང་གྱེས་གྱུར་ན། གཉེན་བཤེས་ཚོགས་ལ་ཇི་ལྟར་ཆགས། ཚུལ་འདི་འདི་ལྟར་མི་བསམ་པར། མི་ཆོས་ཆོས་བརྒྱད་ཁོལ་བོ་རང་། བྱས་ན་ན་རྒ་འཆི་བདག་ཁོ། བྱུང་དུས་དུས་དེར་འགྱོད་པ་སྐྱེ། དེ་ཕྱིར་ཤེས་ལྡན་སྙིང་གི་གྲོགས། ཁྱོད་ལ་མང་པོ་སྨྲ་ཅི་དགོས། མདོར་ན་ཚེ་འདི་བློ་ཡིས་ཐོངས། རི་ཁྲོད་འགྲིམས་ལ་ཉམས་ལེན་མཛོད། བདག་བློ་ཉི་མའི་དཀྱིལ་འཁོར་ལས། ཐོན་པའི་ལེགས་བཤད་འོད་ཟེར་འདི། སྙིང་གྲོགས་བློ་ཡི་པདྨོར་ཞུགས། དགའ་སྤྲོའི་འཛུམ་དགོད་དབང་མེད་འཆོར། བསྒྲུབ་པའི་བརྩོན་འགྲུས་མདངས་སྟོབས་རྒྱས། སྙན་གྲགས་དྲི་ཡིས་ཕྱོགས་ཀུན་ཁྱབ། ཉམས་རྟོགས་ཡོན་ཏན་བཅུད་ཀྱིས་གཏམས། སྐུ་གསུམ་འབྲས་བུ་སྨིན་པར་ཤོག ། ཅེས་སྨྲས་པའོ། བགྲང་ཡས་རྒྱལ་བ་སྲས་བཅས་ཀུན་ལས་འགྲོ་བ་ལ། ཆེས་ཆེར་བརྩེ་བས་གཟིགས་མཛད་པདྨ་འབྱུང་གནས་ཞེས། ཞིང་འདིར་སྐལ་ལྡན་གདུལ་བྱ་ཡོངས་ཀྱི་གཙུག་རྒྱན་དུ། མི་ཡི་རྣམ་རོལ་ཆོས་རྒྱལ་ཆེན་པོར་སྤྱི་བོས་འདུད། མགོན་དེའི་ཐུགས་ཀྱི་ཤེལ་དཀར་བུམ་བཟང་ལས་ཐོན་པའི། དམ་ཆོས་བདུད་རྩིའི་རྒྱུན་གྱིས་རང་རིག་ནོར་བུ་ཡི།

རང་སེམས་ཐུལ་ན་དེ་ཀ་ཡིན། དང་པོ་ངེས་འབྱུང་བློ་ཡིས་ཐུལ། བར་དུ་བྱང་ཆུབ་སེམས་ཀྱིས་ཐུལ། ཐ་མ་བསྐྱེད་རྫོགས་ལམ་གྱིས་ཐུལ། རང་རྒྱུད་ཆོས་དང་ཅི་འདྲེས་མཛོད། གཞན་ཕན་བསམ་སྦྱོར་མི་དམན་འཚལ། དེ་སྐད་པདྨ་བི་ཛ་ཡིས། རྐེ་རྒྱུད་སྨན་མདའི་སྒར་དུ་བྲིས། གཡུང་དྲུང་དགྱེས་པའི་དགའ་སྟོན་མཛོད། དོན་གཉིས་བསམ་པ་ཀུན་འགྲུབ་ཤོག༑༄༅། འབྲོང་བ་རྡོར་ཤུལ་སྟོད་མའི་དཔོན་པོའི་བཞེད་སྐོང་དུ། ཨོཾ་སྭསྟི། ཆོས་དབྱིངས་མཁའ་ལ་མཁྱེན་བརྩེའི་ཆུ་འཛིན་འཁྲིགས། ཐུགས་རྗེའི་ཆར་གྱིས་རྣམ་གྲོལ་མྱུ་གུའི་ཚོགས། སྐྱེད་མཛད་དྲིན་ཆེན་བླ་མར་ཕྱག་བྱས་ནས། ཁྱེད་གསུང་མི་ལྡོག་བསླབ་བྱ་མདོ་ཙམ་བྲི། ངོ་བོ་རྒྱུ་རྐྱེན་དཔེ་གྲངས་གང་བསམས་ཀྱང་། ཤིན་ཏུ་རྙེད་དཀའ་ཡིད་བཞིན་ནོར་འདྲའི་རྟེན། སྔོན་བསགས་བཟང་པོས་ལན་གཅིག་ཐོབ་དུས་འདིར། ཆོས་མིན་བྱ་བས་རང་མགོ་མི་བསྐོར་འཚལ། ཕྱི་སྣོད་འཇིག་རྟེན་དུས་བཞིའི་འགྱུར་ལྡོག་ངོམ། ནང་བཅུད་འགྲོ་ཀུན་སྐྱེ་རྒ་ན་འཆིའི་གཙེས། གསང་བ་ཕུང་ལྔའི་ཚོགས་པ་རྒུད་ཅིང་རྙིང་། ཚུལ་འདིར་དགོངས་ལ་ངེས་འབྱུང་བློ་སྣ་སྟུངས། མ་རིག་གཟུང་འཛིན་ལྕགས་སྒྲོག་དམ་པོ་ཡིས། ཐོག་མེད་དུས་ནས་སྲིད་པའི་བཙོན་རར་བཅིངས། སྡུག་བསྔལ་འགྱུར་དང་འདུ་བྱེད་ཆད་པས་མནར། ད་དུང་ཐར་པའི་སྐབས་མེད་སྐྱོ་བ་སྐྱེད། དཔེར་ན་ས་བོན་མངར་ཁ་གང་བཏབ་པར། འབྲས་བུ་མི་འཁྱོལ་དེ་དེར་སྨིན་པ་བཞིན། ནམ་ཡང་མི་བསླུ་རྒྱུ་འབྲས་ཟབ་མོའི་ཚུལ། རྒྱལ་བ་བདེན་པར་སྨྲ་བའི་གསུང་མཛོད་ལགས། ཀུན་སློང་ཆགས་སྡང་རྨོངས་པས་མ་བསླད་པར། འདིར་ཡང་བདེ་ལ་ཕྱི་མར་ཕན་པ་ཡི། ལས་ཀྱི་ལམ་དེ་འབད་པས་སྒྲུབ་བྱ་ཞིང་། དེ་ལས་ལྡོག་རྣམས་དུག་བཞིན་སྤོང་འཚལ་ལོ། བྱམས་དང་སྙིང་རྗེས་འབངས་འཁོར་བུ་བཞིན་སྐྱོངས།
